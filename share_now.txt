interface RoomAlertNotice {
  type: 'locked' | 'full' | 'notFound' | 'general' | 'duplicate';
  title: string;
  message: string;
}

export function App() {
  const [userName, setUserName] = useState<string>(() => {
    return (
      localStorage.getItem('bridgik_username') ||
      localStorage.getItem('bridgit_username') ||
      localStorage.getItem('airbridge_username') ||
      generateRandomName()
    );
  });
  const [userAvatar, setUserAvatar] = useState<string>(() => {
    const saved = localStorage.getItem('bridgik_user_avatar');
    if (saved) return saved;
    const randomAv = getRandomAvatar();
    localStorage.setItem('bridgik_user_avatar', randomAv);
    return randomAv;
  });
  const [deviceType] = useState(detectDeviceType());
  const [theme, setTheme] = useState<ThemeMode>(() => {
    return (localStorage.getItem('airbridge_theme') as ThemeMode) || 'swiss';
  });

  // Admin & Creator State
  const [isAdmin, setIsAdmin] = useState(false);
  const [clientIp, setClientIp] = useState<string>('');
  const [adminFrame, setAdminFrame] = useState<'gold' | 'rgb' | 'plasma' | 'none'>(() => {
    return (localStorage.getItem('bridgik_admin_frame') as any) || 'gold';
  });
  const [statusEmote, setStatusEmote] = useState<string | null>(() => {
    return localStorage.getItem('bridgik_user_status_emote') || null;
  });

  useEffect(() => {
    const params = new URLSearchParams(window.location.search);
    const urlAdminKey = params.get('admin');
    if (urlAdminKey) {
      localStorage.setItem('bridgik_admin_key', urlAdminKey);
      window.history.replaceState({}, document.title, window.location.pathname);
    }
  }, []);

  const [socket, setSocket] = useState<Socket | null>(null);
  const [isConnected, setIsConnected] = useState(false);
  const [roomId, setRoomId] = useState<string | null>(null);
  const [pin, setPin] = useState<string>('');
  const [activeTab, setActiveTab] = useState<AppTab>('bridge');
  const [legalDoc, setLegalDoc] = useState<LegalDocType>('rules');
  const [roomMode, setRoomMode] = useState<RoomMode>('bridge');
  const [scratchpadText, setScratchpadText] = useState<string>('');
  const [myPeerId, setMyPeerId] = useState<string>('');
  const [peers, setPeers] = useState<Peer[]>([]);
  const [transfers, setTransfers] = useState<FileTransfer[]>([]);
  const [messages, setMessages] = useState<TextMessage[]>([]);
  const [isQRModalOpen, setIsQRModalOpen] = useState(false);
  const [isLoading, setIsLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | undefined>();

  // Local Wi-Fi Discovery (Radar) & Auto-Download
  const [nearbyPeers, setNearbyPeers] = useState<NearbyPeer[]>([]);
  const [incomingPairRequest, setIncomingPairRequest] = useState<PairRequest | null>(null);
  const [isPairing, setIsPairing] = useState(false);
  const [pairingTargetName, setPairingTargetName] = useState<string>('');
  const [pairingTargetSocketId, setPairingTargetSocketId] = useState<string>('');
  const [pairRejectionNotice, setPairRejectionNotice] = useState<string | null>(null);
  const [roomAlertNotice, setRoomAlertNotice] = useState<RoomAlertNotice | null>(null);
  const [participantNotice, setParticipantNotice] = useState<{
    type: 'joined' | 'left';
    name: string;
  } | null>(null);
  const participantNoticeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const peersRef = useRef<Peer[]>(peers);
  useEffect(() => {
    peersRef.current = peers;
  }, [peers]);

  const [lanIp, setLanIp] = useState<string | null>(null);
  const [destinationDirName, setDestinationDirName] = useState<string | null>(null);
  const destinationDirHandleRef = useRef<any>(null);
  const [autoDownload, setAutoDownload] = useState<boolean>(() => {
    return (
      localStorage.getItem('bridgik_auto_download') !== 'false' &&
      localStorage.getItem('bridgit_auto_download') !== 'false'
    );
  });

  // PWA Share Target queue
  const [pendingSharedFiles, setPendingSharedFiles] = useState<File[]>([]);
  const [pendingSharedText, setPendingSharedText] = useState<string>('');
  const pendingShareOpenRoomRef = useRef(false);

  // Media, Voice Activity, Per-peer Volumes & Reactions
  const [isMicActive, setIsMicActive] = useState(false);
  const [localAudioStream, setLocalAudioStream] = useState<MediaStream | null>(null);
  const [speakingPeers, setSpeakingPeers] = useState<Record<string, boolean>>({});
  const [peerVolumes, setPeerVolumes] = useState<Record<string, number>>({});
  const [seatReactions, setSeatReactions] = useState<
    Array<{ id: string; peerId: string; emoteUrl: string }>
  >([]);

  const [isScreenActive, setIsScreenActive] = useState(false);
  const [hasRemoteVideo, setHasRemoteVideo] = useState(false);
  const [roomTheme, setRoomTheme] = useState<string>(() => {
    return localStorage.getItem('bridgik_pixel_room') || 'cyber-lounge';
  });
  const [maxSeats, setMaxSeats] = useState<number>(8);
  const [isLocked, setIsLocked] = useState<boolean>(false);
  const [knockRequired, setKnockRequired] = useState<boolean>(false);
  const [waitingKnock, setWaitingKnock] = useState<{ roomId: string; pin: string } | null>(null);
  const [incomingKnocks, setIncomingKnocks] = useState<
    Array<{ socketId: string; roomId: string; peerName: string }>
  >([]);
  const joinRoomRef = useRef<(pinOrId: string, mode?: RoomMode) => void>(() => {});
  const [peerSeats, setPeerSeats] = useState<Record<string, number>>({});
  const [mySeatIndex, setMySeatIndex] = useState<number>(0);
  const remoteVideoRef = useRef<HTMLVideoElement | null>(null);
  const [remoteAudioStreams, setRemoteAudioStreams] = useState<Record<string, MediaStream>>({});
  const [audioAutoplayBlocked, setAudioAutoplayBlocked] = useState(false);

  const autoDownloadRef = useRef<boolean>(autoDownload);
  useEffect(() => {
    autoDownloadRef.current = autoDownload;
  }, [autoDownload]);

  const isMicActiveRef = useRef<boolean>(isMicActive);
  useEffect(() => {
    isMicActiveRef.current = isMicActive;
  }, [isMicActive]);

  const toggleMicRef = useRef<() => Promise<boolean>>(async () => false);

  // Register Service Worker & read any files shared via Android Share Target
  useEffect(() => {
    if (typeof navigator !== 'undefined' && 'serviceWorker' in navigator) {
      navigator.serviceWorker.register('/sw.js').catch(() => {});
    }

    const params = new URLSearchParams(window.location.search);
    if (params.get('shared') === '1' && 'caches' in window) {
      window.history.replaceState({}, document.title, window.location.pathname);
      (async () => {
        try {
          const cache = await caches.open('bridgik-share-v1');
          const metaResp = await cache.match('/__shared_meta__');
          if (!metaResp) return;

          const meta = await metaResp.json();
          const loadedFiles: File[] = [];
          if (Array.isArray(meta.files)) {
            for (const fMeta of meta.files) {
              const fileResp = await cache.match(fMeta.key);
              if (fileResp) {
                const blob = await fileResp.blob();
                loadedFiles.push(
                  new File([blob], fMeta.name || 'shared-file', {
                    type: fMeta.type || blob.type || 'application/octet-stream',
                  })
                );
              }
            }
          }

          const combinedText = [meta.title, meta.text, meta.url]
            .map((s: any) => String(s || '').trim())
            .filter(Boolean)
            .join('\n');

          await caches.delete('bridgik-share-v1');

          if (loadedFiles.length > 0 || combinedText) {
            setPendingSharedFiles(loadedFiles);
            setPendingSharedText(combinedText);
            pendingShareOpenRoomRef.current = true;
          }
        } catch (err) {
          console.warn('Failed to load shared target cache', err);
        }
      })();
    }
  }, []);

  // Monitor local microphone real RMS volume for speaking indicator
  useEffect(() => {
    if (!localAudioStream || !myPeerId || !isMicActive) {
      if (myPeerId) {
        setSpeakingPeers((prev) => (prev[myPeerId] ? { ...prev, [myPeerId]: false } : prev));
      }
      return;
    }
    const stop = monitorStreamVoiceActivity(localAudioStream, (speaking) => {
      setSpeakingPeers((prev) =>
        prev[myPeerId] === speaking ? prev : { ...prev, [myPeerId]: speaking }
      );
    });
    return () => {
      stop();
      setSpeakingPeers((prev) => (prev[myPeerId] ? { ...prev, [myPeerId]: false } : prev));
    };
  }, [localAudioStream, myPeerId, isMicActive]);

  // Auto-unlock audio & WebAudio AudioContext on first user interaction
  useEffect(() => {
    if (!audioAutoplayBlocked) return;
    const unlockAudio = async () => {
      const ctx = getSharedAudioContext();
      if (ctx && ctx.state === 'suspended') {
        await ctx.resume().catch(() => {});
      }
      const els = Array.from(
        document.querySelectorAll('audio[data-peer-audio="true"]')
      ) as HTMLAudioElement[];
      const results = await Promise.allSettled(
        els.map((audio) => (audio.paused ? audio.play() : Promise.resolve()))
      );
      const hasRejected = results.some((r) => r.status === 'rejected');
      if (!hasRejected && (!ctx || ctx.state === 'running')) {
        setAudioAutoplayBlocked(false);
      }
    };
    window.addEventListener('pointerdown', unlockAudio);
    window.addEventListener('keydown', unlockAudio);
    return () => {
      window.removeEventListener('pointerdown', unlockAudio);
      window.removeEventListener('keydown', unlockAudio);
    };
  }, [audioAutoplayBlocked]);

  const [scratchpadTypingUser, setScratchpadTypingUser] = useState<string | null>(null);
  const scratchpadTypingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const [peerConnectionTypes, setPeerConnectionTypes] = useState<
    Record<string, 'direct' | 'relay'>
  >({});

  const webrtcRef = useRef<WebRTCManager | null>(null);

  const pushSeatReaction = (peerId: string, emoteUrl: string) => {
    const id = `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`;
    setSeatReactions((prev) => [...prev.slice(-15), { id, peerId, emoteUrl }]);
    setTimeout(() => {
      setSeatReactions((prev) => prev.filter((r) => r.id !== id));
    }, 2500);
  };

  // Initialize Socket.io
  useEffect(() => {
    const socketInstance = io({
      transports: ['websocket', 'polling'],
    });

    socketInstance.on('connect', () => {
      setIsConnected(true);
      console.log('Connected to signaling server:', socketInstance.id);
    });

    socketInstance.on('disconnect', () => {
      setIsConnected(false);
    });

    setSocket(socketInstance);

    return () => {
      socketInstance.disconnect();
    };
  }, []);

  // Auto-open a Bridge when arriving via Share Target
  useEffect(() => {
    if (
      socket &&
      isConnected &&
      !roomId &&
      pendingShareOpenRoomRef.current &&
      (pendingSharedFiles.length > 0 || pendingSharedText)
    ) {
      pendingShareOpenRoomRef.current = false;
      handleCreateRoom('bridge');
    }
  }, [socket, isConnected, roomId, pendingSharedFiles, pendingSharedText]);

  useEffect(() => {
    fetch('/api/health')
      .then((r) => r.json())
      .then((data) => {
        if (data?.lanIp && data.lanIp !== '127.0.0.1') {
          setLanIp(data.lanIp);
        }
      })
      .catch(() => {});
  }, []);

  useEffect(() => {
    if (!pairRejectionNotice) return;
    const timer = setTimeout(() => {
      setPairRejectionNotice(null);
    }, 6000);
    return () => clearTimeout(timer);
  }, [pairRejectionNotice]);

  useEffect(() => {
    if (!roomAlertNotice) return;
    const timer = setTimeout(() => {
      setRoomAlertNotice(null);
    }, 7000);
    return () => clearTimeout(timer);
  }, [roomAlertNotice]);

  const handleUpdateUserName = (name: string) => {
    setUserName(name);
    localStorage.setItem('bridgik_username', name);
    if (webrtcRef.current) {
      webrtcRef.current.updateMyName(name);
    }
    setPeers((prev) => prev.map((p) => (p.id === myPeerId ? { ...p, name } : p)));
    if (socket && roomId) {
      socket.emit('update-peer-name', { roomId, name });
    }
    if (socket && !roomId) {
      socket.emit('lobby-update-name', { name });
    }
  };

  const handleUpdateUserAvatar = (avatar: string) => {
    setUserAvatar(avatar);
    localStorage.setItem('bridgik_user_avatar', avatar);
    if (webrtcRef.current) {
      webrtcRef.current.updateMyAvatar(avatar);
    }
    setPeers((prev) => prev.map((p) => (p.id === myPeerId ? { ...p, avatar } : p)));
    setMessages((prev) =>
      prev.map((m) => (m.senderId === myPeerId ? { ...m, senderAvatar: avatar } : m))
    );
    if (socket && roomId) {
      socket.emit('update-peer-avatar', { roomId, avatar, adminFrame });
    }
  };

  const handleUpdateAdminFrame = (frame: 'gold' | 'rgb' | 'plasma' | 'none') => {
    setAdminFrame(frame);
    localStorage.setItem('bridgik_admin_frame', frame);
    webrtcRef.current?.updateAdminStatus(isAdmin, frame);
    setPeers((prev) => prev.map((p) => (p.id === myPeerId ? { ...p, adminFrame: frame } : p)));
    if (socket && roomId) {
      socket.emit('update-peer-avatar', { roomId, avatar: userAvatar, adminFrame: frame });
    }
  };

  const handleUpdateStatusEmote = (emote: string | null) => {
    setStatusEmote(emote);
    if (emote) {
      localStorage.setItem('bridgik_user_status_emote', emote);
    } else {
      localStorage.removeItem('bridgik_user_status_emote');
    }
    if (webrtcRef.current) {
      webrtcRef.current.updateStatusEmote(emote || undefined);
    }
    setPeers((prev) =>
      prev.map((p) => (p.id === myPeerId ? { ...p, statusEmote: emote || undefined } : p))
    );
    setMessages((prev) =>
      prev.map((m) =>
        m.senderId === myPeerId ? { ...m, senderStatusEmote: emote || undefined } : m
      )
    );
    if (socket && roomId) {
      socket.emit('update-peer-status-emote', { roomId, statusEmote: emote || undefined });
    }
    if (socket && !roomId) {
      socket.emit('lobby-update-status-emote', { statusEmote: emote || undefined });
    }
  };

  const handleToggleTheme = () => {
    const nextTheme: ThemeMode =
      theme === 'swiss' ? 'cyberpunk' : theme === 'cyberpunk' ? 'lofi' : 'swiss';
    setTheme(nextTheme);
    localStorage.setItem('airbridge_theme', nextTheme);
  };

  const setupWebRTC = (sock: Socket, myId: string) => {
    if (webrtcRef.current) {
      webrtcRef.current.destroy();
    }

    const manager = new WebRTCManager(
      sock,
      myId,
      userName,
      {
        onTextMessage: (msg) => {
          if (msg.senderId !== myId) {
            sound.playMessage();
          }
          setMessages((prev) => [...prev, msg]);
        },
        onTransferProgress: (updated) => {
          setTransfers((prev) => {
            const index = prev.findIndex((t) => t.id === updated.id);
            if (index >= 0) {
              const next = [...prev];
              next[index] = {
                ...next[index],
                ...updated,
                blobUrl: updated.blobUrl || next[index].blobUrl,
              } as FileTransfer;
              return next;
            } else {
              return [updated as FileTransfer, ...prev];
            }
          });
        },
        onTransferComplete: (completed) => {
          sound.playTransferComplete();

          if (completed.type === 'download' && completed.blobUrl && !completed.savedToDisk) {
            if (autoDownloadRef.current) {
              try {
                const a = document.createElement('a');
                a.href = completed.blobUrl;
                a.download = completed.name;
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
              } catch (err) {
                console.warn('Auto download error', err);
              }
            }
          }

          setTransfers((prev) => {
            const index = prev.findIndex((t) => t.id === completed.id);
            if (index >= 0) {
              const next = [...prev];
              next[index] = {
                ...completed,
                blobUrl: completed.blobUrl || next[index].blobUrl,
              };
              return next;
            }
            return [completed, ...prev];
          });
        },
        onRemoteStream: (peerId, stream) => {
          const videoTracks = stream.getVideoTracks();
          const audioTracks = stream.getAudioTracks();

          if (videoTracks.length > 0 && remoteVideoRef.current) {
            remoteVideoRef.current.srcObject = stream;
            setHasRemoteVideo(true);
            videoTracks[0].onended = () => {
              setHasRemoteVideo(false);
            };
          }

          if (audioTracks.length > 0) {
            setRemoteAudioStreams((prev) => ({ ...prev, [peerId]: stream }));
          }
        },
        onLocalStreamChange: (stream) => {
          setLocalAudioStream(stream);
        },
        onReaction: (peerId, emoteUrl) => {
          sound.playClick();
          pushSeatReaction(peerId, emoteUrl);
        },
        onConnectionFailed: (_peerId, reason) => {
          if (reason === 'NAT_BLOCKED') {
            setRoomAlertNotice({
              type: 'general',
              title: 'Прямое соединение заблокировано сетью',
              message:
                'Файрвол или мобильный оператор блокирует прямой P2P-канал между устройствами.',
            });
          }
        },
        onMemoryLimitWarning: (fileName, size) => {
          const mb = Math.round(size / (1024 * 1024));
          setRoomAlertNotice({
            type: 'general',
            title: 'Крупный файл для мобильного браузера',
            message: `Файл «${fileName}» (${mb} МБ) загружается в оперативную память, так как браузер не поддерживает прямую запись на диск.`,
          });
        },
        onPeerConnectionStateChange: (peerId, state) => {
          console.log(`Peer ${peerId} state changed to ${state}`);
        },
        onScratchpadSync: (text) => {
          setScratchpadText(text);
        },
        onScratchpadTyping: (updatedBy) => {
          setScratchpadTypingUser(updatedBy);
          if (scratchpadTypingTimerRef.current) clearTimeout(scratchpadTypingTimerRef.current);
          scratchpadTypingTimerRef.current = setTimeout(() => {
            setScratchpadTypingUser(null);
          }, 2500);
        },
        onConnectionTypeChange: (peerId, type) => {
          setPeerConnectionTypes((prev) => ({ ...prev, [peerId]: type }));
        },
      },
      userAvatar
    );

    if (destinationDirHandleRef.current) {
      manager.setDestinationDirectory(destinationDirHandleRef.current);
    }

    manager.updateAdminStatus(isAdmin, adminFrame);
    if (statusEmote) {
      manager.updateStatusEmote(statusEmote);
    }
    webrtcRef.current = manager;
    return manager;
  };

  useEffect(() => {
    if (!socket) return;

    socket.on('admin-status', (data: { isAdmin: boolean; clientIp: string }) => {
      setIsAdmin(data.isAdmin);
      setClientIp(data.clientIp);
      webrtcRef.current?.updateAdminStatus(data.isAdmin, adminFrame);
    });

    const storedAdminKey = localStorage.getItem('bridgik_admin_key');
    if (storedAdminKey) {
      socket.emit('verify-admin-key', { key: storedAdminKey });
    }

    socket.on('peer-joined', (newPeer: Peer) => {
      sound.playConnect();
      setPeers((prev) => {
        if (prev.some((p) => p.id === newPeer.id)) return prev;
        return [...prev, newPeer];
      });

      setParticipantNotice({ type: 'joined', name: newPeer.name });
      if (participantNoticeTimerRef.current) clearTimeout(participantNoticeTimerRef.current);
      participantNoticeTimerRef.current = setTimeout(() => setParticipantNotice(null), 4000);

      if (webrtcRef.current) {
        webrtcRef.current.connectToPeer(newPeer.id, true);
      }
    });

    socket.on('peer-left', (data: { peerId: string; peerName?: string }) => {
      const leavingPeer = peersRef.current.find((p) => p.id === data.peerId);
      const peerName = data.peerName || leavingPeer?.name || 'Участник';

      sound.playDisconnect();
      setPeers((prev) => prev.filter((p) => p.id !== data.peerId));
      setRemoteAudioStreams((prev) => {
        const next = { ...prev };
        delete next[data.peerId];
        return next;
      });
      setSpeakingPeers((prev) => {
        const next = { ...prev };
        delete next[data.peerId];
        return next;
      });

      setParticipantNotice({ type: 'left', name: peerName });
      if (participantNoticeTimerRef.current) clearTimeout(participantNoticeTimerRef.current);
      participantNoticeTimerRef.current = setTimeout(() => setParticipantNotice(null), 4000);

      if (webrtcRef.current) {
        webrtcRef.current.removePeer(data.peerId);
      }
    });

    socket.on('peer-updated', (updatedPeer: Peer) => {
      setPeers((prev) =>
        prev.map((p) => (p.id === updatedPeer.id ? { ...p, ...updatedPeer } : p))
      );
      if (updatedPeer.avatar) {
        setMessages((prev) =>
          prev.map((m) =>
            m.senderId === updatedPeer.id ? { ...m, senderAvatar: updatedPeer.avatar } : m
          )
        );
      }
      if (updatedPeer.statusEmote !== undefined) {
        setMessages((prev) =>
          prev.map((m) =>
            m.senderId === updatedPeer.id
              ? { ...m, senderStatusEmote: updatedPeer.statusEmote }
              : m
          )
        );
      }
    });

    socket.on('peer-mic-changed', (data: { peerId: string; isMicActive: boolean }) => {
      setPeers((prev) =>
        prev.map((p) => (p.id === data.peerId ? { ...p, isMicActive: data.isMicActive } : p))
      );
      if (!data.isMicActive) {
        setSpeakingPeers((prev) => ({ ...prev, [data.peerId]: false }));
      }
    });

    socket.on('room-theme-changed', (data: { theme: string }) => {
      if (data.theme) {
        setRoomTheme(data.theme);
        localStorage.setItem('bridgik_pixel_room', data.theme);
      }
    });

    socket.on(
      'room-config-updated',
      (data: { maxSeats?: number; isLocked?: boolean; knockRequired?: boolean }) => {
        if (data.maxSeats !== undefined) setMaxSeats(data.maxSeats);
        if (data.isLocked !== undefined) setIsLocked(data.isLocked);
        if (data.knockRequired !== undefined) setKnockRequired(data.knockRequired);
      }
    );

    socket.on('peer-seat-changed', (data: { peerId: string; seatIndex: number }) => {
      setPeerSeats((prev) => {
        if (data.seatIndex === -1) {
          const next = { ...prev };
          delete next[data.peerId];
          return next;
        }
        return { ...prev, [data.peerId]: data.seatIndex };
      });
    });

    socket.on('kicked', (data: { reason?: string }) => {
      alert(data?.reason || 'Организатор исключил вас из комнаты.');
      handleLeaveRoom();
    });

    socket.on('remote-mute', (_data: { from: string }) => {
      if (isMicActiveRef.current) {
        toggleMicRef.current();
      }
    });

    socket.on('knock-request', (req: { socketId: string; roomId: string; peerName: string }) => {
      sound.playMessage();
      setIncomingKnocks((prev) => {
        if (prev.some((k) => k.socketId === req.socketId)) return prev;
        return [...prev, req];
      });
    });

    socket.on('knock-cancelled', (data: { socketId: string }) => {
      setIncomingKnocks((prev) => prev.filter((k) => k.socketId !== data.socketId));
    });

    socket.on('knock-result', (data: { approved: boolean; roomId: string }) => {
      setWaitingKnock(null);
      if (data.approved) {
        sound.playConnect();
        joinRoomRef.current(data.roomId);
      } else {
        sound.playClick();
        setRoomAlertNotice({
          type: 'locked',
          title: 'Вход отклонён',
          message: 'Организатор отклонил ваш запрос или закрыл комнату.',
        });
      }
    });

    socket.on('room-destroyed', (data: { reason?: string }) => {
      sound.playClick();
      if (webrtcRef.current) {
        webrtcRef.current.destroy();
        webrtcRef.current = null;
      }
      setRoomId(null);
      setPin('');
      setMyPeerId('');
      setPeers([]);
      setTransfers((prev) => {
        prev.forEach((t) => t.blobUrl && URL.revokeObjectURL(t.blobUrl));
        return [];
      });
      setMessages([]);
      setScratchpadText('');
      setIsMicActive(false);
      setLocalAudioStream(null);
      setSpeakingPeers({});
      setSeatReactions([]);
      setIsScreenActive(false);
      setHasRemoteVideo(false);
      setRemoteAudioStreams({});
      setAudioAutoplayBlocked(false);
      setWaitingKnock(null);
      setIncomingKnocks([]);
      setKnockRequired(false);
      setIsLocked(false);
      setMaxSeats(8);
      setPeerSeats({});
      setMySeatIndex(0);
      window.history.replaceState({}, '', window.location.pathname);
      setRoomAlertNotice({
        type: 'locked',
        title: 'Комната уничтожена',
        message: data?.reason || 'Организатор экстренно завершил встречу.',
      });
    });

    socket.on('signal', ({ from, payload }: { from: string; payload: any }) => {
      if (webrtcRef.current) {
        webrtcRef.current.handleSignal(from, payload);
      }
    });

    return () => {
      socket.off('admin-status');
      socket.off('peer-joined');
      socket.off('peer-left');
      socket.off('peer-updated');
      socket.off('peer-mic-changed');
      socket.off('room-theme-changed');
      socket.off('room-config-updated');
      socket.off('peer-seat-changed');
      socket.off('kicked');
      socket.off('remote-mute');
      socket.off('knock-request');
      socket.off('knock-cancelled');
      socket.off('knock-result');
      socket.off('room-destroyed');
      socket.off('signal');
    };
  }, [socket]);

  useEffect(() => {
    if (!socket || !isConnected || roomId) return;

    socket.emit('lobby-join', {
      name: userName,
      deviceType,
      statusEmote: statusEmote || undefined,
    });
    socket.emit('lobby-query');

    const handleNearby = (peers: NearbyPeer[]) => {
      setNearbyPeers(peers);
    };

    const handleIncomingPair = (req: PairRequest) => {
      sound.playMessage();
      setIncomingPairRequest(req);
    };

    const handlePairCancelled = () => {
      setIncomingPairRequest(null);
    };

    const handlePairResult = (res: {
      roomId: string;
      pin: string;
      accepted: boolean;
      reason?: string;
    }) => {
      setIsPairing(false);
      setPairingTargetSocketId('');
      setPairingTargetName('');
      if (res.accepted) {
        sound.playConnect();
        handleJoinRoom(res.roomId, 'bridge');
      } else {
        const reason = res.reason || 'Устройство отклонило запрос на передачу файлов';
        setPairRejectionNotice(reason);
        sound.playClick();
      }
    };

    socket.on('nearby-peers', handleNearby);
    socket.on('incoming-pair-request', handleIncomingPair);
    socket.on('pair-request-cancelled', handlePairCancelled);
    socket.on('direct-pair-result', handlePairResult);

    return () => {
      socket.emit('lobby-leave');
      socket.off('nearby-peers', handleNearby);
      socket.off('incoming-pair-request', handleIncomingPair);
      socket.off('pair-request-cancelled', handlePairCancelled);
      socket.off('direct-pair-result', handlePairResult);
    };
  }, [socket, isConnected, roomId, userName, deviceType]);

  useEffect(() => {
    if (!socket || !isConnected) return;
    const params = new URLSearchParams(window.location.search);
    const roomParam = params.get('room');
    const modeParam = params.get('mode') as RoomMode | null;
    if (roomParam && !roomId) {
      handleJoinRoom(roomParam, modeParam || undefined);
    }
  }, [socket, isConnected]);

  const handleCreateRoom = (mode: RoomMode = 'bridge') => {
    if (!socket) return;
    setIsLoading(true);
    setErrorMessage(undefined);
    setRoomMode(mode);
    setActiveTab(mode);

    socket.emit(
      'create-room',
      {
        peerName: userName,
        avatar: userAvatar,
        adminFrame,
        statusEmote: statusEmote || undefined,
        deviceType,
        mode,
        maxSeats: 8,
        roomTheme,
        browserId: getBrowserDeviceId(),
      },
      (res: any) => {
        setIsLoading(false);
        if (res.success) {
          setRoomId(res.roomId);
          setPin(res.pin);
          setMyPeerId(res.peerId);
          setPeers(res.peers || []);
          if (res.maxSeats !== undefined) setMaxSeats(res.maxSeats);
          if (res.isLocked !== undefined) setIsLocked(res.isLocked);
          if (res.knockRequired !== undefined) setKnockRequired(res.knockRequired);
          if (res.currentRoomTheme) {
            setRoomTheme(res.currentRoomTheme);
          }
          if (res.peerSeats) setPeerSeats(res.peerSeats);
          if (res.mySeat !== undefined) setMySeatIndex(res.mySeat);
          setupWebRTC(socket, res.peerId);
          const modeQuery = mode === 'meeting' ? '&mode=meeting' : '';
          const targetRoom = res.roomId || res.pin;
          window.history.pushState({}, '', `?room=${targetRoom}${modeQuery}`);
        } else {
          setErrorMessage(res.error || 'Не удалось создать комнату');
        }
      }
    );
  };

  const handlePairWithNearby = (targetSocketId: string) => {
    if (!socket) return;
    const targetPeer = nearbyPeers.find((p) => p.socketId === targetSocketId);
    const targetName = targetPeer?.name || 'Устройство';
    setPairingTargetName(targetName);
    setPairingTargetSocketId(targetSocketId);
    setIsPairing(true);
    setErrorMessage(undefined);
    setPairRejectionNotice(null);

    socket.emit('direct-pair-request', {
      toSocketId: targetSocketId,
      myName: userName,
      myDevice: deviceType,
    });
  };

  const handleCancelPairRequest = () => {
    if (!socket) return;
    if (pairingTargetSocketId) {
      socket.emit('cancel-pair-request', { toSocketId: pairingTargetSocketId });
    }
    setIsPairing(false);
    setPairingTargetSocketId('');
    setPairingTargetName('');
  };

  const handleAcceptPair = () => {
    if (!socket || !incomingPairRequest) return;
    const req = incomingPairRequest;
    sound.playConnect();
    socket.emit('direct-pair-response', {
      toSocketId: req.fromSocketId,
      roomId: req.roomId,
      pin: req.pin,
      accepted: true,
    });
    setIncomingPairRequest(null);
    handleJoinRoom(req.roomId, 'bridge');
  };

  const handleRejectPair = () => {
    if (!socket || !incomingPairRequest) return;
    const req = incomingPairRequest;
    sound.playClick();
    socket.emit('direct-pair-response', {
      toSocketId: req.fromSocketId,
      roomId: req.roomId,
      pin: req.pin,
      accepted: false,
      reason: 'Устройство отклонило запрос на передачу файлов',
    });
    setIncomingPairRequest(null);
  };

  const handleToggleAutoDownload = () => {
    setAutoDownload((prev) => {
      const next = !prev;
      localStorage.setItem('bridgik_auto_download', String(next));
      return next;
    });
  };

  const handleCreateMeeting = () => {
    handleCreateRoom('meeting');
  };

  const handleJoinRoom = (pinOrId: string, mode?: RoomMode) => {
    if (!socket) return;
    setIsLoading(true);
    setErrorMessage(undefined);
    const targetMode = mode || (activeTab === 'meeting' ? 'meeting' : 'bridge');
    setRoomMode(targetMode);
    setActiveTab(targetMode);

    socket.emit(
      'join-room',
      {
        roomIdOrPin: pinOrId,
        peerName: userName,
        avatar: userAvatar,
        adminFrame,
        statusEmote: statusEmote || undefined,
        deviceType,
        browserId: getBrowserDeviceId(),
      },
      (res: any) => {
        setIsLoading(false);
        if (res.success) {
          setRoomId(res.roomId);
          setPin(res.pin);
          setMyPeerId(res.peerId);
          setPeers(res.peers || []);
          if (res.maxSeats !== undefined) setMaxSeats(res.maxSeats);
          if (res.isLocked !== undefined) setIsLocked(res.isLocked);
          if (res.knockRequired !== undefined) setKnockRequired(res.knockRequired);
          const resolvedMode = (res.mode as RoomMode) || targetMode;
          setRoomMode(resolvedMode);
          setActiveTab(resolvedMode);
          if (res.currentRoomTheme) {
            setRoomTheme(res.currentRoomTheme);
            localStorage.setItem('bridgik_pixel_room', res.currentRoomTheme);
          }
          if (res.peerSeats) setPeerSeats(res.peerSeats);
          if (res.mySeat !== undefined) setMySeatIndex(res.mySeat);
          setupWebRTC(socket, res.peerId);
          const modeQuery = resolvedMode === 'meeting' ? '&mode=meeting' : '';
          const targetRoom = res.roomId || res.pin;
          window.history.pushState({}, '', `?room=${targetRoom}${modeQuery}`);
        } else {
          sound.playClick();
          window.history.replaceState({}, '', window.location.pathname);
          const errText = res.error || 'Комната не найдена';
          if (res.code === 'KNOCK_REQUIRED' && res.roomId) {
            setWaitingKnock({ roomId: res.roomId, pin: res.pin || pinOrId });
            socket.emit('knock', { roomId: res.roomId, peerName: userName });
            return;
          }
          if (
            res.code === 'ALREADY_IN_ROOM_SAME_BROWSER' ||
            errText.includes('другой вкладке')
          ) {
            setRoomAlertNotice({
              type: 'duplicate',
              title: 'Комната уже открыта',
              message:
                errText ||
                'Вы уже находитесь в этой комнате в другой вкладке браузера. Чтобы подключить второго участника, откройте окно Инкогнито (Ctrl+Shift+N) или другой браузер.',
            });
          } else if (res.code === 'RATE_LIMITED') {
            setRoomAlertNotice({
              type: 'locked',
              title: 'Слишком много попыток',
              message: errText,
            });
          } else if (
            res.code === 'ROOM_LOCKED' ||
            errText.includes('закрыт') ||
            errText.includes('заблокирован')
          ) {
            setRoomAlertNotice({
              type: 'locked',
              title: 'Комната закрыта',
              message: 'Организатор заблокировал вход в эту комнату. Подключение невозможно.',
            });
          } else if (
            res.code === 'ROOM_FULL' ||
            errText.includes('лимит') ||
            errText.includes('переполн')
          ) {
            setRoomAlertNotice({
              type: 'full',
              title: 'Комната переполнена',
              message: errText || 'В комнате достигнут лимит мест. Свободных мест нет.',
            });
          } else {
            setRoomAlertNotice({
              type: 'notFound',
              title: 'Комната не найдена',
              message: errText || 'Комната не найдена или код неверен.',
            });
          }
          setErrorMessage(errText);
        }
      }
    );
  };
  joinRoomRef.current = handleJoinRoom;

  const handlePickDestinationDir = async () => {
    if (typeof window === 'undefined' || !('showDirectoryPicker' in window)) {
      alert(
        'Ваш браузер не поддерживает выбор папки на диске. Используйте Google Chrome, Microsoft Edge или Яндекс Браузер.'
      );
      return;
    }
    try {
      const handle = await (window as any).showDirectoryPicker({
        mode: 'readwrite',
      });
      destinationDirHandleRef.current = handle;
      setDestinationDirName(handle.name);
      if (webrtcRef.current) {
        webrtcRef.current.setDestinationDirectory(handle);
      }
    } catch (err: any) {
      if (err.name !== 'AbortError') {
        console.warn('showDirectoryPicker error', err);
      }
    }
  };

  const handleClearDestinationDir = () => {
    destinationDirHandleRef.current = null;
    setDestinationDirName(null);
    if (webrtcRef.current) {
      webrtcRef.current.setDestinationDirectory(null);
    }
  };

  const handleLeaveRoom = () => {
    if (socket) {
      socket.emit('leave-room');
    }
    if (webrtcRef.current) {
      webrtcRef.current.destroy();
      webrtcRef.current = null;
    }
    setRoomId(null);
    setPin('');
    setMyPeerId('');
    setPeers([]);
    setTransfers((prev) => {
      prev.forEach((t) => {
        if (t.blobUrl) URL.revokeObjectURL(t.blobUrl);
      });
      return [];
    });
    setMessages([]);
    setScratchpadText('');
    setAudioAutoplayBlocked(false);
    setIsMicActive(false);
    setLocalAudioStream(null);
    setSpeakingPeers({});
    setSeatReactions([]);
    setIsScreenActive(false);
    setHasRemoteVideo(false);
    setRemoteAudioStreams({});
    setMaxSeats(8);
    setIsLocked(false);
    setKnockRequired(false);
    setIncomingKnocks([]);
    setPeerSeats({});
    setMySeatIndex(0);
    window.history.replaceState({}, '', window.location.pathname);
  };

  const handleUpdateMaxSeats = (newMax: number) => {
    setMaxSeats(newMax);
    if (socket && (pin || roomId)) {
      socket.emit('update-room-config', {
        roomId: pin || roomId,
        maxSeats: newMax,
      });
    }
  };

  const handleToggleLockRoom = () => {
    const nextLocked = !isLocked;
    setIsLocked(nextLocked);
    if (socket && (pin || roomId)) {
      socket.emit('update-room-config', {
        roomId: pin || roomId,
        isLocked: nextLocked,
      });
    }
  };

  const handleToggleKnockRequired = () => {
    const nextKnock = !knockRequired;
    setKnockRequired(nextKnock);
    if (socket && (pin || roomId)) {
      socket.emit('update-room-config', {
        roomId: pin || roomId,
        knockRequired: nextKnock,
      });
    }
  };

  const handleKnockDecision = (
    targetSocketId: string,
    targetRoomId: string,
    approved: boolean
  ) => {
    if (!socket) return;
    sound.playClick();
    socket.emit('knock-decision', {
      roomId: targetRoomId,
      toSocketId: targetSocketId,
      approved,
    });
    setIncomingKnocks((prev) => prev.filter((k) => k.socketId !== targetSocketId));
  };

  const handleCancelKnock = () => {
    if (socket && waitingKnock) {
      socket.emit('cancel-knock', { roomId: waitingKnock.roomId });
    }
    setWaitingKnock(null);
  };

  const handlePanicDestroy = () => {
    const amHost =
      peersRef.current.find((p) => p.id === myPeerId)?.isHost ??
      peersRef.current.length <= 1;
    if (socket && amHost && (pin || roomId)) {
      socket.emit('destroy-room', { roomId: pin || roomId });
    }
    handleLeaveRoom();
    setRoomAlertNotice({
      type: 'locked',
      title: 'Экстренный выход (Паника)',
      message: amHost
        ? 'Комната уничтожена для всех, каналы и локальная память очищены.'
        : 'Все каналы разорваны, локальная история и буфер стёрты.',
    });
  };

  const handleSwitchSeat = (seatIdx: number) => {
    setMySeatIndex(seatIdx);
    if (socket && (pin || roomId)) {
      socket.emit('change-seat', {
        roomId: pin || roomId,
        seatIndex: seatIdx,
      });
    }
  };

  const handleChangePeerVolume = (peerId: string, volumePercent: number) => {
    setPeerVolumes((prev) => ({
      ...prev,
      [peerId]: Math.max(0, Math.min(200, volumePercent)),
    }));
  };

  const handleSendReaction = (emoteUrl: string, emoteName?: string) => {
    sound.playClick();
    if (myPeerId) {
      pushSeatReaction(myPeerId, emoteUrl);
    }
    webrtcRef.current?.sendReaction(emoteUrl, emoteName);
  };

  const handleSendFiles = (files: FileList | File[]) => {
    if (!webrtcRef.current) return;
    const fileArray = Array.from(files);
    fileArray.forEach((file) => {
      webrtcRef.current?.sendFile(file);
    });
  };

  const handleSendText = (text: string, isClipboard = false) => {
    if (!webrtcRef.current) return;
    webrtcRef.current.sendTextMessage(text, isClipboard);
  };

  const handleSendSharedBundle = () => {
    if (pendingSharedFiles.length > 0) {
      handleSendFiles(pendingSharedFiles);
    }
    if (pendingSharedText) {
      handleSendText(pendingSharedText, true);
    }
    setPendingSharedFiles([]);
    setPendingSharedText('');
  };

  const handleDismissSharedBundle = () => {
    setPendingSharedFiles([]);
    setPendingSharedText('');
  };

  const handleUpdateScratchpad = (text: string) => {
    setScratchpadText(text);
    if (webrtcRef.current) {
      webrtcRef.current.sendScratchpadUpdate(text);
      webrtcRef.current.sendScratchpadTyping();
    }
  };

  const handleCancelTransfer = (id: string) => {
    if (!webrtcRef.current) return;
    webrtcRef.current.cancelTransfer(id);
  };

  const handleAcceptFile = async (fileId: string) => {
    if (webrtcRef.current) {
      await webrtcRef.current.acceptIncomingFile(fileId);
    }
  };

  const handleRejectFile = (fileId: string) => {
    if (webrtcRef.current) {
      webrtcRef.current.rejectIncomingFile(fileId);
    }
  };

  const handleToggleMic = async () => {
    if (!webrtcRef.current) {
      sound.playClick();
      setRoomAlertNotice({
        type: 'general',
        title: 'Микрофон',
        message: 'Комната инициализируется, подождите 1-2 секунды и нажмите снова.',
      });
      return false;
    }

    try {
      const active = await webrtcRef.current.toggleMicrophone();
      setIsMicActive(active);
      sound.playClick();
      setPeers((prev) =>
        prev.map((p) => (p.id === myPeerId ? { ...p, isMicActive: active } : p))
      );
      if (socket && roomId) {
        socket.emit('toggle-mic', { roomId, isMicActive: active });
      }
      return active;
    } catch (err: any) {
      sound.playClick();
      console.error('Microphone toggle error:', err);
      setRoomAlertNotice({
        type: 'general',
        title: 'Ошибка микрофона',
        message: err.message || 'Не удалось получить доступ к микрофону.',
      });
      setIsMicActive(false);
      return false;
    }
  };
  toggleMicRef.current = handleToggleMic;

  const handleToggleScreen = async () => {
    if (!webrtcRef.current) return false;
    const active = await webrtcRef.current.toggleScreenShare();
    setIsScreenActive(active);
    return active;
  };

  const roomUrl = (() => {
    if (typeof window === 'undefined') return '';
    const isLocalhost =
      window.location.hostname === 'localhost' || window.location.hostname === '127.0.0.1';
    const host = isLocalhost && lanIp ? `${lanIp}:${window.location.port}` : window.location.host;
    const modeQuery = roomMode === 'meeting' ? '&mode=meeting' : '';
    const targetRoom = roomId || pin;
    return `${window.location.protocol}//${host}?room=${targetRoom}${modeQuery}`;
  })();

  const getThemeWrapperClass = () => {
    return 'bg-[#171717] text-[#f2f2f2] selection:bg-[#789d2a] selection:text-white';
  };

  return (
    <div
      className={`min-h-screen flex flex-col overflow-x-hidden transition-colors ${getThemeWrapperClass()}`}
    >
      <Navbar
        userName={userName}
        onUpdateUserName={handleUpdateUserName}
        userAvatar={userAvatar}
        onUpdateUserAvatar={handleUpdateUserAvatar}
        isAdmin={isAdmin}
        adminFrame={adminFrame}
        onUpdateAdminFrame={handleUpdateAdminFrame}
        statusEmote={statusEmote}
        onUpdateStatusEmote={handleUpdateStatusEmote}
        clientIp={clientIp}
        deviceType={deviceType}
        theme={theme}
        onToggleTheme={handleToggleTheme}
        isConnected={isConnected}
        onGoHome={roomId ? handleLeaveRoom : () => setActiveTab('bridge')}
        activeTab={activeTab}
        onSelectTab={setActiveTab}
      />

      <main className="flex-1">
        {activeTab === 'legal' ? (
          <LegalView onBack={() => setActiveTab('bridge')} defaultDoc={legalDoc} />
        ) : activeTab === 'about' ? (
          <AboutView onBack={() => setActiveTab('bridge')} />
        ) : !roomId ? (
          <HomeView
            onCreateBridge={() => handleCreateRoom('bridge')}
            onJoinBridge={(pinOrId) => handleJoinRoom(pinOrId, 'bridge')}
            onCreateMeeting={handleCreateMeeting}
            onJoinMeeting={(pinOrId) => handleJoinRoom(pinOrId, 'meeting')}
            activeMode={activeTab === 'meeting' ? 'meeting' : 'bridge'}
            onSelectMode={(mode) => setActiveTab(mode)}
            isLoading={isLoading}
            errorMessage={errorMessage}
            nearbyPeers={nearbyPeers}
            onPairWithNearby={handlePairWithNearby}
            isPairing={isPairing}
            incomingPairRequest={incomingPairRequest}
            onAcceptPair={handleAcceptPair}
            onRejectPair={handleRejectPair}
            myUserName={userName}
            myDeviceType={deviceType}
            lanIp={lanIp}
            onOpenAbout={() => setActiveTab('about')}
          />
        ) : roomMode === 'meeting' ? (
          <MeetingRoomView
            socket={socket}
            roomId={roomId}
            pin={pin}
            myPeerId={myPeerId}
            myUserName={userName}
            onUpdateUserName={handleUpdateUserName}
            userAvatar={userAvatar}
            onUpdateUserAvatar={handleUpdateUserAvatar}
            isAdmin={isAdmin}
            adminFrame={adminFrame}
            onUpdateAdminFrame={handleUpdateAdminFrame}
            statusEmote={statusEmote}
            onUpdateStatusEmote={handleUpdateStatusEmote}
            clientIp={clientIp}
            roomTheme={roomTheme}
            onRoomThemeChange={(theme) => setRoomTheme(theme)}
            maxSeats={maxSeats}
            onUpdateMaxSeats={handleUpdateMaxSeats}
            isLocked={isLocked}
            onToggleLockRoom={handleToggleLockRoom}
            knockRequired={knockRequired}
            onToggleKnockRequired={handleToggleKnockRequired}
            onPanicDestroy={handlePanicDestroy}
            peerSeats={peerSeats}
            mySeatIndex={mySeatIndex}
            onSwitchSeat={handleSwitchSeat}
            speakingPeers={speakingPeers}
            peerVolumes={peerVolumes}
            onChangePeerVolume={handleChangePeerVolume}
            seatReactions={seatReactions}
            onSendReaction={handleSendReaction}
            roomUrl={roomUrl}
            peers={peers}
            peerConnectionTypes={peerConnectionTypes}
            transfers={transfers}
            messages={messages}
            scratchpadText={scratchpadText}
            scratchpadTypingUser={scratchpadTypingUser}
            onUpdateScratchpad={handleUpdateScratchpad}
            onSendFiles={handleSendFiles}
            onSendText={handleSendText}
            onLeaveRoom={handleLeaveRoom}
            onOpenQR={() => setIsQRModalOpen(true)}
            onToggleMic={handleToggleMic}
            onToggleScreen={handleToggleScreen}
            remoteVideoRef={remoteVideoRef}
            isMicActive={isMicActive}
            isScreenActive={isScreenActive}
            hasRemoteVideo={hasRemoteVideo}
            onAcceptFile={handleAcceptFile}
          />

function RemotePeerAudio({
  stream,
  volumePercent = 100,
  onSpeakingChange,
  onAutoplayBlocked,
}: {
  stream: MediaStream;
  volumePercent?: number;
  onSpeakingChange?: (speaking: boolean) => void;
  onAutoplayBlocked?: () => void;
}) {
  const audioRef = useRef<HTMLAudioElement>(null);
  const gainNodeRef = useRef<GainNode | null>(null);
  const isIOS =
    typeof navigator !== 'undefined' && /iPad|iPhone|iPod/.test(navigator.userAgent);

  useEffect(() => {
    if (gainNodeRef.current) {
      gainNodeRef.current.gain.value = Math.max(0, Math.min(2, volumePercent / 100));
    } else if (audioRef.current) {
      audioRef.current.volume = Math.max(0, Math.min(1, volumePercent / 100));
    }
  }, [volumePercent]);

  useEffect(() => {
    const audio = audioRef.current;
    if (!audio || !stream) return;

    audio.srcObject = stream;

    const ctx = !isIOS ? getSharedAudioContext() : null;
    let source: MediaStreamAudioSourceNode | null = null;
    let gainNode: GainNode | null = null;
    let analyser: AnalyserNode | null = null;
    let rafId = 0;
    let stopped = false;
    let stopMonitor: (() => void) | null = null;

    if (ctx) {
      try {
        // Mute native element so Web Audio GainNode handles 0-200% output cleanly without echo
        audio.muted = true;
        source = ctx.createMediaStreamSource(stream);
        gainNode = ctx.createGain();
        gainNode.gain.value = Math.max(0, Math.min(2, volumePercent / 100));
        gainNodeRef.current = gainNode;

        analyser = ctx.createAnalyser();
        analyser.fftSize = 256;

        source.connect(gainNode);
        gainNode.connect(ctx.destination);
        source.connect(analyser);

        const data = new Uint8Array(analyser.fftSize);
        let lastSpokeAt = 0;
        let isSpeaking = false;

        const tick = () => {
          if (stopped || !analyser) return;
          analyser.getByteTimeDomainData(data);
          let sum = 0;
          for (let i = 0; i < data.length; i++) {
            const norm = (data[i] - 128) / 128;
            sum += norm * norm;
          }
          const rms = Math.sqrt(sum / data.length);
          const now = performance.now();
          if (rms > 0.015) {
            lastSpokeAt = now;
            if (!isSpeaking) {
              isSpeaking = true;
              onSpeakingChange?.(true);
            }
          } else if (isSpeaking && now - lastSpokeAt > 300) {
            isSpeaking = false;
            onSpeakingChange?.(false);
          }
          rafId = requestAnimationFrame(tick);
        };
        rafId = requestAnimationFrame(tick);
      } catch {
        gainNodeRef.current = null;
        audio.muted = false;
        audio.volume = Math.max(0, Math.min(1, volumePercent / 100));
      }
    } else {
      audio.muted = false;
      audio.volume = Math.max(0, Math.min(1, volumePercent / 100));
      stopMonitor = monitorStreamVoiceActivity(stream, (speaking) => {
        onSpeakingChange?.(speaking);
      });
    }

    audio.play().catch((err) => {
      if (err?.name !== 'AbortError') {
        onAutoplayBlocked?.();
      }
    });

    const timer = setTimeout(() => {
      if (audio.paused || (ctx && ctx.state === 'suspended')) {
        onAutoplayBlocked?.();
      }
    }, 300);

    return () => {
      stopped = true;
      clearTimeout(timer);
      if (rafId) cancelAnimationFrame(rafId);
      stopMonitor?.();
      try {
        source?.disconnect();
        gainNode?.disconnect();
        analyser?.disconnect();
      } catch {}
      gainNodeRef.current = null;
      onSpeakingChange?.(false);
    };
  }, [stream]);

  return <audio ref={audioRef} data-peer-audio="true" autoPlay playsInline />;
}

  const handleOpenChatTab = () => {
    setActiveSideTab('chat');
    setUnreadChatCount(0);
    setRecentlyReceivedBadge(null);
    setNewChatToast(null);
  };

  useEffect(() => {
    const handleDragEnter = (e: DragEvent) => {
      e.preventDefault();
      dragCounter.current += 1;
      if (e.dataTransfer && Array.from(e.dataTransfer.types).includes('Files')) {
        setIsGlobalDragging(true);
      }
    };

    const handleDragLeave = (e: DragEvent) => {
      e.preventDefault();
      dragCounter.current -= 1;
      if (dragCounter.current <= 0) {
        dragCounter.current = 0;
        setIsGlobalDragging(false);
      }
    };

    const handleDragOver = (e: DragEvent) => {
      e.preventDefault();
      if (e.dataTransfer) {
        e.dataTransfer.dropEffect = 'copy';
      }
    };

    const handleDrop = (e: DragEvent) => {
      e.preventDefault();
      dragCounter.current = 0;
      setIsGlobalDragging(false);
      if (e.dataTransfer?.files && e.dataTransfer.files.length > 0) {
        sound.playConnect();
        onSendFiles(e.dataTransfer.files);
        setFileToast(`Добавлено файлов: ${e.dataTransfer.files.length}`);
        setTimeout(() => setFileToast(null), 3500);
      }
    };

    const handlePaste = (e: ClipboardEvent) => {
      if (e.clipboardData?.files && e.clipboardData.files.length > 0) {
        e.preventDefault();
        sound.playConnect();
        onSendFiles(e.clipboardData.files);
        setFileToast('Файл из буфера добавлен в файлы комнаты');
        setTimeout(() => setFileToast(null), 3500);
      }
    };

    window.addEventListener('dragenter', handleDragEnter);
    window.addEventListener('dragleave', handleDragLeave);
    window.addEventListener('dragover', handleDragOver);
    window.addEventListener('drop', handleDrop);
    window.addEventListener('paste', handlePaste);

    return () => {
      window.removeEventListener('dragenter', handleDragEnter);
      window.removeEventListener('dragleave', handleDragLeave);
      window.removeEventListener('dragover', handleDragOver);
      window.removeEventListener('drop', handleDrop);
      window.removeEventListener('paste', handlePaste);
    };
  }, [onSendFiles]);

  const handleSwitchSeat = (seatIdx: number) => {
    if (seatIdx === mySeatIndex) return;
    sound.playClick();
    onSwitchSeat(seatIdx);
  };

  const handleUpdateMaxSeats = (newMax: number) => {
    sound.playClick();
    onUpdateMaxSeats(newMax);
  };

  const handleToggleLockRoom = () => {
    sound.playClick();
    onToggleLockRoom();
  };

  const handleKickPeer = (targetPeerId: string) => {
    if (!confirm('Вы уверены, что хотите исключить этого участника из комнаты?')) return;
    sound.playClick();
    if (socket) {
      socket.emit('kick-peer', {
        roomId: pin || _roomId,
        targetPeerId,
      });
    }
  };

  const handleMuteAll = () => {
    sound.playClick();
    if (socket) {
      socket.emit('mute-all', {
        roomId: pin || _roomId,
      });
    }
  };

  useEffect(() => {
    if (myUserName && !customNameInput) {
      setCustomNameInput(myUserName);
    }
  }, [myUserName]);

  useEffect(() => {
    const timer = setInterval(() => {
      setMeetingSeconds((prev) => prev + 1);
    }, 1000);
    return () => clearInterval(timer);
  }, []);

  const formatTimer = (seconds: number) => {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  };

  useEffect(() => {
    chatBottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleCopyLink = () => {
    navigator.clipboard.writeText(roomUrl);
    setCopiedLink(true);
    sound.playClick();
    setTimeout(() => setCopiedLink(false), 2000);
  };

  const handleCopyNotes = () => {
    navigator.clipboard.writeText(scratchpadText);
    setCopiedNotes(true);
    sound.playClick();
    setTimeout(() => setCopiedNotes(false), 2000);
  };

  const handleDownloadMarkdown = () => {
    const blob = new Blob([scratchpadText], { type: 'text/markdown;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `meeting-notes-${pin}.md`;
    a.click();
    URL.revokeObjectURL(url);
    sound.playClick();
  };

  const handleSendChat = (e: React.FormEvent) => {
    e.preventDefault();
    if (chatInput.trim()) {
      onSendText(chatInput.trim());
      setChatInput('');
    }
  };

  const handleSelectRoom = (roomId: RoomId) => {
    if (!isHost) return;
    setCurrentRoomId(roomId);
    localStorage.setItem('bridgik_pixel_room', roomId);
    setIsRoomPickerOpen(false);
    sound.playClick();
    if (socket) {
      socket.emit('change-room-theme', {
        roomId: pin || _roomId,
        theme: roomId,
      });
    }
    if (onRoomThemeChange) {
      onRoomThemeChange(roomId);
    }
  };

  const handleToggleSound = () => {
    const next = !soundEnabled;
    setSoundEnabled(next);
    sound.setEnabled(next);
    if (next) sound.playClick();
  };

  const handleSaveName = () => {
    const trimmed = customNameInput.trim();
    if (trimmed && onUpdateUserName) {
      onUpdateUserName(trimmed);
      setNameSavedSuccess(true);
      sound.playClick();
      setTimeout(() => setNameSavedSuccess(false), 2000);
    }
  };

  const otherPeers = peers.filter((p) => p.id !== myPeerId);
  const currentRoom = ROOM_PRESETS[currentRoomId] || ROOM_PRESETS['cyber-lounge'];
  const activeSeats = currentRoom.seats.slice(0, maxSeats);

  const actualMySeat = mySeatIndex >= 0 && mySeatIndex < maxSeats ? mySeatIndex : 0;
  const seatOccupants: Record<
    number,
    { type: 'me' } | { type: 'peer'; peer: (typeof otherPeers)[0] }
  > = {
    [actualMySeat]: { type: 'me' },
  };

  const occupiedIndices = new Set<number>([actualMySeat]);

  otherPeers.forEach((peer) => {
    const preferred = peerSeats[peer.id];
    if (
      preferred !== undefined &&
      preferred >= 0 &&
      preferred < maxSeats &&
      !occupiedIndices.has(preferred)
    ) {
      seatOccupants[preferred] = { type: 'peer', peer };
      occupiedIndices.add(preferred);
    }
  });

  otherPeers.forEach((peer) => {
    if (Object.values(seatOccupants).some((o) => o.type === 'peer' && o.peer.id === peer.id)) {
      return;
    }
    for (let i = 0; i < maxSeats; i++) {
      if (!occupiedIndices.has(i)) {
        seatOccupants[i] = { type: 'peer', peer };
        occupiedIndices.add(i);
        break;
      }
    }
  });

  return (
    <div className="max-w-7xl mx-auto px-3 sm:px-4 py-4 sm:py-5">
      {/* Top Header Bar */}
      <div className="bg-[#1e1e1e] border border-[#353535] rounded-xl p-3 sm:p-4 mb-4 sm:mb-5 flex flex-wrap items-center justify-between gap-2.5 sm:gap-3 shadow-md">
        <div className="flex items-center gap-2 sm:gap-3 flex-wrap">
          <div className="flex items-center gap-2 bg-[#141414] px-3 py-1.5 rounded-lg border border-[#333333]">
            <span className="text-xs text-[#888888] uppercase font-bold tracking-wider">PIN:</span>
            <span className="font-mono text-xl sm:text-2xl font-extrabold tracking-wider text-[#9ac43e]">
              {pin}
            </span>
          </div>

          <div className="flex items-center gap-2 px-2.5 sm:px-3 py-1.5 rounded-lg bg-[#141414] border border-[#333333] text-xs sm:text-sm font-mono">
            <span className="w-2 h-2 rounded-full bg-[#789d2a] animate-pulse" />
            <span className="text-[#ededed] font-bold">{formatTimer(meetingSeconds)}</span>
          </div>

          <div className="flex items-center gap-1.5 sm:gap-2 text-xs sm:text-sm text-[#aaaaaa]">
            <Users className="w-4 h-4 text-[#9ac43e]" />
            <span className="font-semibold text-[#ededed]">
              {peers.length} / {maxSeats} {maxSeats === 2 ? 'места' : 'мест'}
            </span>
          </div>

          {isHost && (
            <div className="flex items-center gap-1.5 px-2.5 py-1 rounded-lg bg-[#f59e0b]/15 border border-[#f59e0b]/30 text-xs font-bold text-[#f59e0b]">
              <Crown className="w-3.5 h-3.5" />
              <span>Вы организатор</span>
            </div>
          )}

          {knockRequired && (
            <div className="flex items-center gap-1.5 px-2.5 py-1 rounded-lg bg-[#789d2a]/15 border border-[#789d2a]/40 text-xs font-bold text-[#bbf348]">
              <Bell className="w-3.5 h-3.5" />
              <span>Зал ожидания</span>
            </div>
          )}

          {isLocked && (
            <div className="flex items-center gap-1.5 px-2.5 py-1 rounded-lg bg-[#ed2b3b]/15 border border-[#ed2b3b]/30 text-xs font-bold text-[#ef9282]">
              <Lock className="w-3.5 h-3.5" />
              <span>Комната закрыта</span>
            </div>
          )}
        </div>

        {/* Header Right Actions */}
        <div className="flex items-center gap-2 flex-wrap">
          {isHost ? (
            <button
              onClick={() => {
                setIsRoomPickerOpen(!isRoomPickerOpen);
                sound.playClick();
              }}
              className={`px-3 py-1.5 rounded-lg border text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer ${currentRoom.badgeBg} ${currentRoom.badgeBorder} ${currentRoom.badgeText} hover:brightness-110`}
              title="Выбрать другую локацию (Организатор)"
            >
              <Home className="w-3.5 h-3.5" />
              <span>Локация: {currentRoom.title}</span>
            </button>
          ) : (
            <div
              className={`px-3 py-1.5 rounded-lg border text-xs sm:text-sm font-bold flex items-center gap-1.5 ${currentRoom.badgeBg} ${currentRoom.badgeBorder} ${currentRoom.badgeText}`}
              title="Локация установлена организатором"
            >
              <Home className="w-3.5 h-3.5" />
              <span>Локация: {currentRoom.title}</span>
            </div>
          )}

          <button
            onClick={() => {
              setIsSettingsOpen(true);
              sound.playClick();
            }}
            className="p-2 rounded-lg bg-[#272727] hover:bg-[#333333] text-[#ededed] border border-[#3a3a3a] text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
            title="Настройки комнаты"
          >
            <Settings className="w-4 h-4 text-[#9ac43e]" />
          </button>
        </div>
      </div>

      {/* Main Grid */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 sm:gap-6">
        {/* Left Column */}
        <div className="lg:col-span-7 space-y-5">
          {/* Living Room Stage */}
          <div className="relative rounded-3xl border-2 border-[#33333e] bg-[#0c0c12] shadow-2xl overflow-hidden flex flex-col items-center justify-center">
            <div className="absolute top-3 left-3 right-3 z-20 flex items-center justify-between gap-2 pointer-events-none">
              <div className="flex items-center gap-1.5 px-3 py-1 rounded-xl bg-[#0c0c14]/90 border border-white/10 shadow-lg backdrop-blur-xs pointer-events-auto">
                <span
                  className="w-2 h-2 rounded-full animate-pulse"
                  style={{ backgroundColor: currentRoom.accent }}
                />
                <span className="text-xs font-mono font-extrabold uppercase tracking-wider text-[#ededed]">
                  {currentRoom.title}
                </span>
              </div>

              <div className="flex items-center gap-2 pointer-events-auto">
                <div className="flex items-center gap-1.5 px-2.5 py-1 rounded-xl bg-[#0c0c14]/90 border border-white/10 shadow-lg backdrop-blur-xs text-[11px] font-mono text-[#aaaaaa]">
                  <Armchair className="w-3.5 h-3.5 text-[#9ac43e]" />
                  <span>{maxSeats} мест · P2P Direct</span>
                </div>
              </div>
            </div>

            <div className="relative w-full aspect-video max-h-[520px] overflow-hidden flex items-center justify-center">
              <img
                src={currentRoom.imageSrc}
                alt={currentRoom.title}
                className="w-full h-full object-cover select-none pointer-events-none"
              />

              <div
                className="absolute inset-0 pointer-events-none opacity-20 shadow-[inset_0_0_80px_rgba(0,0,0,0.8)]"
                style={{
                  backgroundImage:
                    'linear-gradient(rgba(18, 16, 16, 0) 50%, rgba(0, 0, 0, 0.3) 50%)',
                  backgroundSize: '100% 4px',
                }}
              />

              {activeSeats.map((seat, index) => {
                const occupant = seatOccupants[index];
                let seatContent: React.ReactNode = null;

                if (occupant?.type === 'me') {
                  seatContent = (
                    <PixelAvatarSeat
                      name={activeMyName}
                      avatar={userAvatar || getAvatarForName(activeMyName)}
                      isMe={true}
                      deviceType={myPeer?.deviceType}
                      isMicActive={isMicActive}
                      spotName={seat.spotName}
                      accentColor={currentRoom.accent}
                      onToggleMic={onToggleMic}
                      isHost={isHost}
                      isAdmin={isAdmin}
                      adminFrame={adminFrame}
                      statusEmote={statusEmote || undefined}
                    />
                  );
                } else if (occupant?.type === 'peer') {
                  const peer = occupant.peer;
                  seatContent = (
                    <PixelAvatarSeat
                      name={peer.name}
                      avatar={peer.avatar || getAvatarForName(peer.name)}
                      isMe={false}
                      deviceType={peer.deviceType}
                      isMicActive={Boolean(peer.isMicActive)}
                      spotName={seat.spotName}
                      accentColor={currentRoom.accent}
                      isHost={peer.isHost}
                      isAdmin={peer.isAdmin}
                      adminFrame={peer.adminFrame}
                      statusEmote={peer.statusEmote}
                      connectionType={peerConnectionTypes?.[peer.id]}
                      onKick={isHost ? () => handleKickPeer(peer.id) : undefined}
                    />
                  );
                } else {
                  seatContent = (
                    <EmptyPixelSeat
                      spotName={seat.spotName}
                      accentColor={currentRoom.accent}
                      onTakeSeat={() => handleSwitchSeat(index)}
                    />
                  );
                }

                return (
                  <div
                    key={seat.id}
                    className="absolute z-10 -translate-x-1/2 -translate-y-1/2 transition-all duration-300"
                    style={{
                      left: `${seat.x}%`,
                      top: `${seat.y}%`,
                    }}
                  >
                    {seatContent}
                  </div>
                );
              })}
            </div>

            {/* Control Dock (below room on mobile, floating HUD on desktop) */}
            <div className="relative sm:absolute sm:bottom-3 sm:left-1/2 sm:-translate-x-1/2 z-30 flex flex-wrap sm:flex-nowrap items-center justify-center gap-1.5 sm:gap-2 px-2.5 sm:px-3 py-2 w-full sm:w-auto sm:max-w-[96%] bg-[#121218]/95 border-t sm:border border-white/15 sm:rounded-2xl shadow-2xl backdrop-blur-md">
              <button
                onClick={onToggleMic}
                className={`px-3 sm:px-3.5 py-1.5 rounded-xl text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-all cursor-pointer ${
                  isMicActive
                    ? 'bg-[#ed2b3b] hover:bg-[#d92231] text-white shadow-[0_0_12px_rgba(237,43,59,0.5)]'
                    : 'bg-[#25252c] hover:bg-[#303038] text-[#ededed] border border-white/10'
                }`}
                title={isMicActive ? 'Заглушить микрофон' : 'Включить микрофон'}
              >
                {isMicActive ? (
                  <Mic className="w-4 h-4 animate-pulse" />
                ) : (
                  <MicOff className="w-4 h-4" />
                )}
                <span>{isMicActive ? 'Микрофон вкл' : 'Микрофон'}</span>
              </button>

              {isHost && (
                <button
                  onClick={() => {
                    setIsRoomPickerOpen(!isRoomPickerOpen);
                    sound.playClick();
                  }}
                  className="px-2.5 py-1.5 rounded-xl bg-[#25252c] hover:bg-[#303038] text-[#ededed] border border-white/10 text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
                  title="Выбрать другую локацию (Организатор)"
                >
                  <Home className="w-4 h-4 text-[#9ac43e]" />
                  <span className="hidden md:inline">Локация</span>
                </button>
              )}

              <button
                onClick={() => {
                  meetingFilesSectionRef.current?.scrollIntoView({ behavior: 'smooth' });
                }}
                className="px-2.5 py-1.5 rounded-xl bg-[#25252c] hover:bg-[#303038] text-[#ededed] border border-white/10 text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
                title="Файлы комнаты"
              >
                <Paperclip className="w-4 h-4 text-[#9ac43e]" />
                <span className="hidden md:inline">Файлы</span>
                {transfers.length > 0 && (
                  <span className="px-1.5 py-0.2 rounded-full bg-[#789d2a] text-white text-[10px] font-mono font-bold">
                    {transfers.length}
                  </span>
                )}
              </button>

              <button
                onClick={onOpenQR}
                className="p-1.5 sm:px-2.5 sm:py-1.5 rounded-xl bg-[#25252c] hover:bg-[#303038] text-[#ededed] border border-white/10 text-xs sm:text-sm font-bold flex items-center gap-1 transition-colors cursor-pointer"
                title="Показать QR-код"
              >
                <QrCode className="w-4 h-4 text-[#9ac43e]" />
                <span className="hidden sm:inline">QR</span>
              </button>

              <button
                onClick={handleCopyLink}
                className="px-2.5 py-1.5 rounded-xl bg-[#25252c] hover:bg-[#303038] text-[#ededed] border border-white/10 text-xs sm:text-sm font-bold flex items-center gap-1 transition-colors cursor-pointer"
                title="Скопировать ссылку на комнату"
              >
                {copiedLink ? (
                  <>
                    <Check className="w-4 h-4 text-[#9ac43e]" />
                    <span className="text-[#9ac43e] hidden sm:inline">Скопировано</span>
                  </>
                ) : (

                    <span className="hidden sm:inline">Ссылка</span>
                  </>
                )}
              </button>

                onClick={onLeaveRoom}

                title="Покинуть комнату"
              >
                <LogOut className="w-4 h-4" />
                <span className="hidden sm:inline">Выйти</span>
              </button>

              {onPanicDestroy && (
                <button
                  onClick={onPanicDestroy}
                  className="px-2.5 sm:px-3 py-1.5 rounded-xl bg-[#ed2b3b]/20 hover:bg-[#ed2b3b] text-[#ef9282] hover:text-white border border-[#ed2b3b]/50 text-xs sm:text-sm font-bold flex items-center gap-1 transition-colors cursor-pointer shadow-[0_0_12px_rgba(237,43,59,0.25)]"
                  title={
                    isHost
                      ? 'ПАНИКА: мгновенно уничтожить комнату для всех'
                      : 'ПАНИКА: экстренно всё стереть и выйти'
                  }
                >
                  <AlertTriangle className="w-4 h-4" />
                  <span>Паника</span>
                </button>
              )}
            </div>

            {isRoomPickerOpen && (
              <div className="absolute bottom-14 sm:bottom-16 left-1/2 -translate-x-1/2 z-40 bg-[#14141c] border border-white/20 rounded-2xl p-3 shadow-2xl w-84 max-w-[95%] animate-in fade-in zoom-in-95 duration-150">
                <div className="flex items-center justify-between pb-2 mb-2 border-b border-white/10">
                  <span className="text-xs font-bold text-[#ededed] uppercase font-mono tracking-wider">
                    Выберите локацию (все 8 мест):
                  </span>
                  <button
                    onClick={() => setIsRoomPickerOpen(false)}
                    className="p-1 rounded text-[#888888] hover:text-white"
                  >
                    <X className="w-3.5 h-3.5" />
                  </button>
                </div>

                <div className="space-y-1.5 max-h-60 overflow-y-auto">
                  {Object.values(ROOM_PRESETS).map((r) => {
                    const isSelected = currentRoomId === r.id;
                    return (
                      <button
                        key={r.id}
                        onClick={() => handleSelectRoom(r.id)}
                        className={`w-full p-2 rounded-xl border text-left flex items-center gap-2.5 transition-all cursor-pointer ${
                          isSelected
                            ? 'bg-[#222230] border-[#9ac43e]'
                            : 'bg-[#181820] border-white/5 hover:border-white/20'
                        }`}
                      >
                        <img
                          src={r.imageSrc}
                          alt={r.title}
                          className="w-14 h-9 rounded-lg object-cover border border-white/10 shrink-0"
                        />
                        <div className="min-w-0 flex-1">
                          <p className="text-xs font-extrabold text-[#ededed] truncate">
                            {r.title}
                          </p>
                          <p className="text-[10px] text-[#888888] truncate">{r.subtitle}</p>
                        </div>
                        {isSelected && (
                          <CheckCircle2 className="w-4 h-4 text-[#9ac43e] shrink-0" />
                        )}
                      </button>
                    );
                  })}
                </div>
              </div>
            )}
          </div>

          {/* Room Files Drawer */}
          <div
            ref={meetingFilesSectionRef}
            className="bg-[#1e1e1e] border border-[#353535] rounded-2xl p-4 sm:p-5 shadow-sm scroll-mt-6"
          >
            <div className="flex items-center justify-between mb-3">
              <div className="flex items-center gap-2">
                <Paperclip className="w-4 h-4 text-[#9ac43e]" />
                <h3 className="text-base sm:text-lg font-bold text-[#ededed]">Файлы комнаты</h3>
                {transfers.length > 0 && (
                  <span className="px-2 py-0.5 rounded-full bg-[#272727] text-[#9ac43e] text-xs font-mono font-bold">
                    {transfers.length}
                  </span>
                )}
              </div>
              <button
                onClick={() => fileInputRef.current?.click()}
                className="px-3 py-1.5 rounded-lg bg-[#272727] hover:bg-[#333333] text-[#ededed] text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer border border-[#3a3a3a]"
              >
                <Upload className="w-3.5 h-3.5" />
                <span>Прикрепить файл</span>
              </button>
              <input
                ref={fileInputRef}
                type="file"
                multiple
                className="hidden"
                onChange={(e) => {
                  if (e.target.files && e.target.files.length > 0) {
                    onSendFiles(e.target.files);
                    setFileToast(`Добавлено файлов: ${e.target.files.length}`);
                    setTimeout(() => setFileToast(null), 3500);
                    e.target.value = '';
                  }
                }}
              />
            </div>

            {transfers.length > 0 ? (
              <div className="space-y-2.5 max-h-60 overflow-y-auto pr-1 scrollbar-thin">
                {transfers.map((t) => {
                  const isImg = isImageFile(t.name, t.mimeType);
                  const isAud = isAudioFile(t.name, t.mimeType);
                  const isVid = isVideoFile(t.name, t.mimeType);

                  return (
                    <div
                      key={t.id}
                      className="p-3 rounded-xl bg-[#141416] border border-[#2e2e38] hover:border-[#789d2a]/50 transition-all flex flex-col gap-2 text-sm shadow-sm"
                    >
                      <div className="flex items-center justify-between gap-3">

                          {isImg && t.blobUrl ? (

                                setPreviewMedia({

                          ) : isVid ? (
                            <div
                              onClick={() =>
                                t.blobUrl &&
                                setPreviewMedia({
                                  url: t.blobUrl,
                                  name: t.name,
                                  type: 'video',
                                  size: t.size,
                                })
                              }
                              className={`w-12 h-12 rounded-xl flex items-center justify-center shrink-0 border ${
                                t.blobUrl

                              title={t.blobUrl ? 'Смотреть видео' : 'Видео'}
                            >
                              <Film className="w-5 h-5" />

                            <div
                              className="w-12 h-12 rounded-xl bg-[#172214] border border-[#789d2a]/50 flex items-center justify-center shrink-0 text-[#9ac43e]"

                          ) : (
                            <div className="w-12 h-12 rounded-xl bg-[#1c1c24] border border-[#383844] flex items-center justify-center shrink-0 text-[#aaaaaa]">

                          <div className="min-w-0">
                            <p className="font-bold text-[#ededed] truncate max-w-[170px] sm:max-w-xs">

                                  onClick={() =>
                                    setPreviewMedia({
                                      url: t.blobUrl!,
                                      name: t.name,
                                      type: isImg ? 'image' : isVid ? 'video' : 'audio',
                                      size: t.size,

                                  }
                                  className="p-2 rounded-xl bg-[#25252e] hover:bg-[#30303c] text-[#ededed] hover:text-[#9ac43e] border border-white/10 transition-colors cursor-pointer shadow-sm"
                                  title="Предпросмотр файла"
                                >
                                  <Eye className="w-4 h-4 text-[#9ac43e]" />
                                </button>

                              <a
                                href={t.blobUrl}
                                download={t.name}

                                <Download className="w-4 h-4" />

                              Принять

                  );
                })}

            ) : (
              <div
                onClick={() => fileInputRef.current?.click()}

                <span className="hidden sm:inline">
                  Перетащите сюда документы или файлы для участников комнаты

            )}
          </div>
        </div>

          <div className="flex items-center border-b border-[#353535] bg-[#171717]">
            <button
              onClick={() => setActiveSideTab('notes')}
              className={`flex-1 py-3 text-base font-bold flex items-center justify-center gap-2 transition-colors cursor-pointer border-b-2 ${
                activeSideTab === 'notes'
                  ? 'border-[#789d2a] text-[#ededed] bg-[#1e1e1e]'
                  : 'border-transparent text-[#888888] hover:text-[#ededed]'

            >
              <FileText className="w-4 h-4 text-[#9ac43e]" />

              onClick={handleOpenChatTab}
              className={`flex-1 py-3 text-base font-bold flex items-center justify-center gap-2 transition-colors cursor-pointer border-b-2 relative ${
                activeSideTab === 'chat'
                  ? 'border-[#789d2a] text-[#ededed] bg-[#1e1e1e]'
                  : 'border-transparent text-[#888888] hover:text-[#ededed]'
              }`}
            >
              <div className="relative flex items-center">
                <Send className="w-4 h-4 text-[#9ac43e]" />
                {(unreadChatCount > 0 || recentlyReceivedBadge !== null) && (
                  <span className="absolute -top-1.5 -right-1.5 w-2 h-2 rounded-full bg-[#bbf348] animate-ping" />
                )}
              </div>
              <span>Чат</span>

              {unreadChatCount > 0 || recentlyReceivedBadge !== null ? (
                <span className="px-2 py-0.5 rounded-full bg-[#789d2a] text-white text-xs font-black font-mono shadow-[0_0_12px_rgba(120,157,42,0.9)] animate-bounce flex items-center gap-1 border border-[#bbf348]">
                  +{unreadChatCount > 0 ? unreadChatCount : recentlyReceivedBadge}
                </span>
              ) : (
                <span className="text-xs font-mono text-[#777777]">({messages.length})</span>
              )}
            </button>
          </div>

          {activeSideTab === 'notes' ? (
            <div className="flex-1 flex flex-col p-4">
              <div className="flex items-center justify-between mb-3 text-xs sm:text-sm text-[#aaaaaa]">
                <span className="font-semibold text-[#9ac43e] flex items-center gap-1.5">
                  <span className="w-2 h-2 rounded-full bg-[#789d2a]" />
                  <span>Канвас · Синхронизация на лету</span>
                </span>
                <div className="flex items-center gap-1.5 sm:gap-2">
                  <button
                    onClick={() => fileInputRef.current?.click()}
                    className="p-1.5 hover:text-white rounded bg-[#272727] hover:bg-[#333333] transition-colors cursor-pointer text-xs flex items-center gap-1 border border-[#383838]"
                    title="Прикрепить файл к комнате"
                  >
                    <Paperclip className="w-3.5 h-3.5 text-[#9ac43e]" />
                    <span className="hidden sm:inline">Файл</span>
                  </button>
                  <button
                    onClick={handleCopyNotes}
                    className="p-1.5 hover:text-white rounded bg-[#272727] hover:bg-[#333333] transition-colors cursor-pointer text-xs flex items-center gap-1"
                    title="Скопировать Канвас"
                  >
                    {copiedNotes ? (
                      <Check className="w-3.5 h-3.5 text-[#9ac43e]" />
                    ) : (
                      <Copy className="w-3.5 h-3.5" />
                    )}
                    <span className="hidden sm:inline">
                      {copiedNotes ? 'Скопировано' : 'Копировать'}
                    </span>
                  </button>
                  <button
                    onClick={handleDownloadMarkdown}
                    className="p-1.5 hover:text-white rounded bg-[#272727] hover:bg-[#333333] transition-colors cursor-pointer text-xs flex items-center gap-1"
                    title="Скачать в .md"
                  >
                    <Download className="w-3.5 h-3.5" />
                    <span>.md</span>
                  </button>
                </div>
              </div>

              {scratchpadTypingUser && (
                <div className="flex items-center gap-1.5 text-xs text-[#9ac43e] font-mono animate-pulse mb-1.5 px-1">
                  <span>✏️</span>
                  <span>{scratchpadTypingUser} печатает в Канвасе...</span>
                </div>
              )}

              <textarea
                ref={scratchpadTextareaRef}
                value={scratchpadText}
                onChange={(e) => {
                  const target = e.target;
                  scratchpadSelectionRef.current = {
                    start: target.selectionStart,
                    end: target.selectionEnd,
                  };
                  onUpdateScratchpad(target.value);
                }}
                onSelect={(e) => {
                  const target = e.target as HTMLTextAreaElement;
                  scratchpadSelectionRef.current = {
                    start: target.selectionStart,
                    end: target.selectionEnd,
                  };
                }}
                placeholder="Канвас комнаты... Набросайте план, тезисы, ссылки или код. Все изменения моментально видны собеседникам."
                className="flex-1 w-full bg-[#141414] border border-[#333333] rounded-xl p-3.5 text-base sm:text-lg text-[#ededed] placeholder-[#666666] outline-none focus:border-[#789d2a] resize-none font-mono leading-relaxed transition-colors"
              />

              <div className="mt-2 text-xs text-[#888888] flex items-center justify-between">
                <span>Канвас не сохраняется на сервере (P2P)</span>
                <span>Символов: {scratchpadText.length}</span>
              </div>
            </div>
          ) : (
            <div className="flex-1 flex flex-col min-h-0">
              <div className="flex-1 p-4 overflow-y-auto space-y-3 scrollbar-thin">
                {messages.length === 0 ? (
                  <div className="h-full flex items-center justify-center text-center text-[#777777] text-sm">
                    Сообщений пока нет. Напишите что-нибудь собеседникам в гостиной.
                  </div>
                ) : (
                  messages.map((m) => {
                    const isMe = m.senderId === myPeerId;
                    const peer = peers.find((p) => p.id === m.senderId);
                    const currentAvatar = isMe
                      ? userAvatar || getAvatarForName(activeMyName)
                      : peer?.avatar || m.senderAvatar || getAvatarForName(m.senderName);
                    const messageIsAdmin = isMe ? isAdmin : Boolean(peer?.isAdmin || m.isAdmin);
                    const messageAdminFrame = isMe
                      ? adminFrame
                      : peer?.adminFrame || m.adminFrame || 'gold';
                    return (
                      <ChatMessageBubble
                        key={m.id}
                        senderName={m.senderName}
                        text={m.text}
                        isMe={isMe}
                        isSystem={m.isSystem || m.senderId === 'system'}
                        avatar={currentAvatar}
                        isAdmin={messageIsAdmin}
                        adminFrame={messageAdminFrame}
                        senderStatusEmote={
                          m.senderStatusEmote || (isMe ? statusEmote || undefined : peer?.statusEmote)
                        }
                        time={new Date(m.timestamp).toLocaleTimeString([], {
                          hour: '2-digit',
                          minute: '2-digit',
                        })}
                      />
                    );
                  })
                )}
                <div ref={chatBottomRef} />
              </div>

              <div className="relative border-t border-[#353535] bg-[#171717]">
                <EmotePicker
                  isOpen={isEmotePickerOpen}
                  onClose={() => setIsEmotePickerOpen(false)}
                  onSelectEmote={(emote) => {
                    setChatInput((prev) => (prev ? `${prev} ${emote.code} ` : `${emote.code} `));
                    setIsEmotePickerOpen(false);

                <form onSubmit={handleSendChat} className="p-2.5 sm:p-3 flex items-center gap-1.5 sm:gap-2">
                  <div
                    className="relative shrink-0 select-none"
                    title={isAdmin ? `Создатель: ${activeMyName}` : `Ваш аватар: ${activeMyName}`}
                  >
                    <img
                      src={userAvatar || getAvatarForName(activeMyName)}
                      alt={activeMyName}
                      className={`w-8 h-8 sm:w-10 sm:h-10 rounded-full object-cover bg-[#171717] ${
                        isAdmin && adminFrame !== 'none'
                          ? adminFrame === 'rgb'
                            ? 'border-2 border-sky-400 animate-admin-rgb'
                            : adminFrame === 'plasma'
                            ? 'border-2 border-purple-400 animate-admin-plasma'
                            : 'border-2 border-amber-400 animate-admin-gold'
                          : 'border-2 border-[#789d2a] shadow-[0_0_8px_rgba(120,157,42,0.35)]'

                      style={{ imageRendering: 'pixelated' }}
                    />
                    {isAdmin && adminFrame !== 'none' && (
                      <span className="absolute -top-1.5 -right-1 text-[11px] select-none filter drop-shadow-[0_0_4px_rgba(245,158,11,0.8)]">
                        👑
                      </span>
                    )}
                    <span className="absolute -bottom-0.5 -right-0.5 w-2.5 h-2.5 bg-[#789d2a] rounded-full border-2 border-[#171717]" />

                  <input
                    type="text"
                    value={chatInput}
                    onChange={(e) => setChatInput(e.target.value)}
                    placeholder="Сообщение или ссылка..."
                    className="flex-1 min-w-0 w-full bg-[#141414] border border-[#333333] rounded-xl px-3 sm:px-3.5 py-2 text-sm sm:text-base text-[#ededed] placeholder-[#666666] outline-none focus:border-[#789d2a] transition-colors"
                  />
                  <button
                    type="button"
                    onClick={() => fileInputRef.current?.click()}
                    className="p-2 sm:p-2.5 rounded-xl border bg-[#222222] border-[#333333] text-[#888888] hover:text-[#9ac43e] hover:border-[#789d2a] transition-colors cursor-pointer flex items-center justify-center shrink-0"
                    title="Прикрепить файл к комнате"
                  >
                    <Paperclip className="w-4 h-4 sm:w-5 sm:h-5" />
                  </button>
                  <button
                    type="button"
                    onClick={() => setIsEmotePickerOpen(!isEmotePickerOpen)}
                    className={`p-2 sm:p-2.5 rounded-xl border transition-colors cursor-pointer flex items-center justify-center shrink-0 ${
                      isEmotePickerOpen
                        ? 'bg-[#789d2a]/20 border-[#789d2a] text-[#9ac43e]'
                        : 'bg-[#222222] border-[#333333] text-[#888888] hover:text-[#ededed] hover:border-[#555555]'
                    }`}
                    title="Смайлики (7TV и мемы)"
                  >
                    <Smile className="w-4 h-4 sm:w-5 sm:h-5" />
                  </button>
                  <button
                    type="submit"
                    disabled={!chatInput.trim()}
                    className="px-3 sm:px-4 py-2 sm:py-2.5 bg-[#789d2a] hover:bg-[#8cb632] disabled:opacity-40 text-white rounded-xl font-bold text-sm transition-colors cursor-pointer flex items-center justify-center shrink-0 shadow-md shadow-[#789d2a]/20"
                  >
                    <Send className="w-4 h-4" />
                  </button>
                </form>
              </div>
            </div>
          )}
        </div>
      </div>

      {/* Settings Modal */}
      {isSettingsOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/80 backdrop-blur-xs animate-in fade-in duration-200">
          <div className="bg-[#1b1b1f] border border-[#35353f] rounded-3xl max-w-xl w-full shadow-2xl relative max-h-[88vh] flex flex-col overflow-hidden">
            <div className="flex items-center justify-between p-5 sm:p-6 pb-4 border-b border-[#303038] shrink-0 bg-[#1b1b1f]">
              <div className="flex items-center gap-2.5">
                <div className="w-9 h-9 rounded-xl bg-[#25252e] border border-[#383845] flex items-center justify-center text-[#9ac43e]">
                  <Settings className="w-5 h-5" />
                </div>
                <div>
                  <h3 className="text-xl font-extrabold text-[#ededed]">Настройки гостиной</h3>
                  <p className="text-xs text-[#888888]">
                    Выбор интерьера (все 8 мест), позывной и звук
                  </p>
                </div>
              </div>
              <button
                onClick={() => {
                  setIsSettingsOpen(false);
                  sound.playClick();
                }}
                className="p-2 rounded-lg bg-[#25252e] hover:bg-[#303038] text-[#aaaaaa] hover:text-white transition-colors cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <div className="flex-1 overflow-y-auto px-5 sm:px-6 py-5 space-y-6 [scrollbar-color:#383848_transparent] [scrollbar-width:thin]">
              {/* Host Settings Section */}
              {isHost && (
                <div className="p-4 rounded-2xl bg-[#14141c] border border-[#f59e0b]/30 space-y-4">
                  <div className="flex items-center justify-between">
                    <div className="flex items-center gap-2">
                      <Crown className="w-4 h-4 text-[#f59e0b]" />
                      <h4 className="text-base font-bold text-[#ededed]">Панель организатора</h4>
                    </div>
                    <span className="text-[10px] uppercase font-bold px-2 py-0.5 rounded bg-[#f59e0b]/20 text-[#f59e0b] border border-[#f59e0b]/40">
                      Хост
                    </span>
                  </div>

                  <div>
                    <label className="text-xs font-semibold text-[#aaaaaa] block mb-1.5">
                      Количество мест в комнате (на диванах и креслах):
                    </label>
                    <div className="grid grid-cols-4 gap-2">
                      {[2, 4, 6, 8].map((seatsCount) => (
                        <button
                          key={seatsCount}
                          type="button"
                          onClick={() => handleUpdateMaxSeats(seatsCount)}
                          className={`py-2 px-3 rounded-xl text-xs font-bold transition-all cursor-pointer flex items-center justify-center gap-1.5 ${
                            maxSeats === seatsCount
                              ? 'bg-[#789d2a] text-white shadow-md'
                              : 'bg-[#1e1e28] text-[#cccccc] hover:bg-[#282836] border border-white/5'
                          }`}
                        >
                          <Armchair className="w-3.5 h-3.5" />
                          <span>{seatsCount} мест</span>
                        </button>
                      ))}
                    </div>
                  </div>

                  <div className="flex flex-wrap items-center gap-2 pt-1">
                    <button
                      type="button"
                      onClick={handleToggleLockRoom}
                      className={`flex-1 py-2 px-3 rounded-xl text-xs font-bold flex items-center justify-center gap-2 transition-all cursor-pointer border ${
                        isLocked
                          ? 'bg-[#ed2b3b]/15 text-[#ef9282] border-[#ed2b3b]/40 hover:bg-[#ed2b3b]/25'
                          : 'bg-[#1e1e28] text-[#ededed] border-white/10 hover:bg-[#282836]'
                      }`}
                    >
                      {isLocked ? (
                        <Lock className="w-3.5 h-3.5" />
                      ) : (
                        <Unlock className="w-3.5 h-3.5 text-[#9ac43e]" />
                      )}
                      <span>
                        {isLocked
                          ? 'Комната закрыта (вход заблокирован)'
                          : 'Комната открыта для гостей'}
                      </span>
                    </button>

                    {onToggleKnockRequired && (
                      <button
                        type="button"
                        onClick={() => {
                          sound.playClick();
                          onToggleKnockRequired();
                        }}
                        className={`flex-1 py-2 px-3 rounded-xl text-xs font-bold flex items-center justify-center gap-2 transition-all cursor-pointer border ${
                          knockRequired
                            ? 'bg-[#789d2a]/20 text-[#bbf348] border-[#789d2a]'
                            : 'bg-[#1e1e28] text-[#ededed] border-white/10 hover:bg-[#282836]'
                        }`}
                      >
                        <Bell
                          className={`w-3.5 h-3.5 ${
                            knockRequired ? 'text-[#bbf348]' : 'text-[#888888]'
                          }`}
                        />
                        <span>
                          {knockRequired
                            ? 'Стук в дверь: ВКЛ (по одобрению)'
                            : 'Стук в дверь: ВЫКЛ'}
                        </span>
                      </button>
                    )}

                    <button
                      type="button"
                      onClick={handleMuteAll}
                      className="py-2 px-3 rounded-xl text-xs font-bold flex items-center justify-center gap-1.5 bg-[#1e1e28] hover:bg-[#282836] text-[#ededed] border border-white/10 transition-all cursor-pointer"
                      title="Отключить микрофоны всех участников"
                    >
                      <MicOff className="w-3.5 h-3.5 text-[#ed2b3b]" />
                      <span>Заглушить всех</span>
                    </button>
                  </div>

                  {otherPeers.length > 0 && (
                    <div className="pt-2 border-t border-white/5 space-y-1.5">
                      <span className="text-[11px] font-bold text-[#888888] uppercase tracking-wider">
                        Участники в комнате ({otherPeers.length}):
                      </span>
                      <div className="space-y-1 max-h-36 overflow-y-auto">
                        {otherPeers.map((p) => (
                          <div
                            key={p.id}
                            className="flex items-center justify-between p-2 rounded-xl bg-[#1a1a24] border border-white/5 text-xs"
                          >
                            <span className="font-semibold text-[#ededed] truncate">{p.name}</span>
                            <button
                              type="button"
                              onClick={() => handleKickPeer(p.id)}
                              className="px-2 py-1 rounded-lg bg-[#ed2b3b]/15 hover:bg-[#ed2b3b] text-[#ef9282] hover:text-white text-[11px] font-bold transition-colors flex items-center gap-1 cursor-pointer"
                            >
                              <UserX className="w-3 h-3" />
                              <span>Исключить</span>
                            </button>
                          </div>
                        ))}
                      </div>
                    </div>
                  )}
                </div>
              )}

              {/* Room Location */}
              {isHost ? (
                <div>
                  <div className="flex items-center justify-between mb-3">
                    <div className="flex items-center gap-2">

      } else {
        setToastMessage('Буфер обмена пуст');
      }
    } catch (e) {
      console.warn('Clipboard read permission denied', e);
      setToastMessage('Разрешите доступ к буферу обмена в браузере');
    }
  };

  const handleDrop = async (e: React.DragEvent) => {
    e.preventDefault();
    setIsDragging(false);

    if (e.dataTransfer.items && e.dataTransfer.items.length > 0) {
      try {
        const extracted = await extractFilesFromDataTransfer(e.dataTransfer.items);
        if (extracted.length > 0) {
          onSendFiles(extracted);
          setToastMessage(`Загружено файлов: ${extracted.length}`);
          return;
        }
      } catch (err) {
        console.warn('Folder recursive read fallback', err);
      }
    }

    if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
      onSendFiles(e.dataTransfer.files);
    }
  };

  const getFileIcon = (mimeType: string) => {
    if (mimeType.startsWith('image/')) return <ImageIcon className="w-5 h-5 text-[#9ac43e]" />;
    if (mimeType.startsWith('video/')) return <Video className="w-5 h-5 text-[#5192ec]" />;
    if (mimeType.startsWith('audio/')) return <Music className="w-5 h-5 text-[#e8b33c]" />;
    if (mimeType.includes('pdf') || mimeType.includes('text'))
      return <FileText className="w-5 h-5 text-[#aaaaaa]" />;
    return <File className="w-5 h-5 text-[#999999]" />;
  };

  const isUrl = (text: string) => {
    return text.startsWith('http://') || text.startsWith('https://');
  };

  const otherPeers = peers.filter((p) => p.id !== myPeerId);

  return (
    <div className="max-w-6xl mx-auto px-3 sm:px-4 py-4 sm:py-6">
      {/* Top Bar */}
      <div className="bg-[#222222] border border-[#3a3a3a] rounded-xl p-3.5 sm:p-5 mb-5 sm:mb-6 flex flex-col gap-3 shadow-md">
        <div className="flex flex-wrap items-center justify-between gap-2.5 sm:gap-3">
          <div className="flex items-center gap-2.5 sm:gap-3.5">
            <div className="flex items-center gap-2 bg-[#171717] px-3 sm:px-4 py-1.5 sm:py-2 rounded-lg border border-[#3a3a3a]">
              <span className="text-xs sm:text-sm text-[#999999] uppercase font-bold tracking-wider">
                PIN:
              </span>
              <span className="font-mono text-xl sm:text-2xl font-extrabold tracking-wider text-[#9ac43e]">
                {pin}
              </span>
            </div>

            <div className="flex items-center gap-1.5 sm:gap-2 text-xs sm:text-base text-[#ededed]">
              <Users className="w-4 h-4 text-[#9ac43e] shrink-0" />
              <span className="font-semibold">
                {otherPeers.length === 0
                  ? 'Ожидание участников...'
                  : `${peers.length} устройств(а) в сети`}
              </span>
            </div>
          </div>

          <div className="flex items-center gap-1.5 sm:gap-2 flex-wrap">
            <button
              onClick={onToggleAutoDownload}
              className={`px-2.5 sm:px-3.5 py-1.5 sm:py-2 rounded-lg text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer border ${
                autoDownload
                  ? 'bg-[#1c2214] border-[#789d2a] text-[#9ac43e]'
                  : 'bg-[#2d2d2d] border-[#3a3a3a] text-[#aaaaaa] hover:text-[#ededed]'
              }`}
              title={
                autoDownload
                  ? 'Автосохранение включено: полученные файлы скачиваются сразу'
                  : 'Автосохранение выключено: файлы скачиваются вручную'
              }
            >
              {autoDownload ? (
                <CheckSquare className="w-4 h-4 text-[#9ac43e] shrink-0" />
              ) : (
                <Square className="w-4 h-4 shrink-0" />
              )}
              <span className="sm:hidden">{autoDownload ? 'Авто: ВКЛ' : 'Авто: ВЫКЛ'}</span>
              <span className="hidden sm:inline">
                {autoDownload ? 'Автосохранение' : 'Автосохранение: ВЫКЛ'}
              </span>
            </button>

            {typeof window !== 'undefined' &&
              'showDirectoryPicker' in window &&
              (destinationDirName ? (
                <div className="flex items-center gap-1.5 bg-[#1c2214] border border-[#789d2a] rounded-lg px-2.5 sm:px-3 py-1.5 sm:py-2 text-xs sm:text-sm text-[#9ac43e]">
                  <FolderDown className="w-4 h-4 text-[#9ac43e] shrink-0" />
                  <span
                    className="font-bold max-w-[110px] sm:max-w-[130px] truncate"
                    title={`Папка на диске: ${destinationDirName}`}
                  >
                    Папка: {destinationDirName}
                  </span>
                  <button
                    onClick={onClearDestinationDir}
                    className="p-0.5 hover:text-white transition-colors cursor-pointer"
                    title="Сбросить выбранную папку"
                  >
                    <X className="w-3.5 h-3.5 text-[#aaaaaa] hover:text-[#ef9282]" />
                  </button>
                </div>
              ) : (
                <button
                  onClick={onPickDestinationDir}
                  className="px-2.5 sm:px-3.5 py-1.5 sm:py-2 rounded-lg bg-[#2d2d2d] hover:bg-[#3a3a3a] text-[#ededed] text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer border border-[#3a3a3a]"
                  title="Выберите папку на диске для автоматического сохранения"
                >
                  <FolderDown className="w-4 h-4 text-[#9ac43e]" />
                  <span className="hidden sm:inline">Папка на диске</span>
                </button>
              ))}

            <button
              onClick={onOpenQR}
              className="px-2.5 sm:px-3.5 py-1.5 sm:py-2 rounded-lg bg-[#2d2d2d] hover:bg-[#3a3a3a] text-[#ededed] text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
            >
              <QrCode className="w-4 h-4 text-[#9ac43e]" />
              <span className="sm:hidden">QR</span>
              <span className="hidden sm:inline">QR-код</span>
            </button>

            <button
              onClick={handleCopyLink}
              className="px-2.5 sm:px-3.5 py-1.5 sm:py-2 rounded-lg bg-[#2d2d2d] hover:bg-[#3a3a3a] text-[#ededed] text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
            >
              {copiedLink ? (
                <>
                  <Check className="w-4 h-4 text-[#9ac43e]" />
                  <span className="text-[#9ac43e]">Скопировано</span>
                </>
              ) : (
                <>
                  <Share2 className="w-4 h-4" />
                  <span>Ссылка</span>
                </>
              )}
            </button>

            <button
              onClick={onLeaveRoom}
              className="px-2.5 sm:px-3.5 py-1.5 sm:py-2 rounded-lg bg-[#2d2d2d] hover:bg-[#ef9282]/15 hover:text-[#ef9282] border border-transparent hover:border-[#ef9282]/30 text-[#aaaaaa] text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
              title="Закрыть комнату и очистить память"
            >
              <LogOut className="w-4 h-4" />
              <span>Выйти</span>
            </button>
          </div>
        </div>

        {peers.length > 0 && (
          <div className="flex items-center gap-2 pt-2.5 sm:pt-3 border-t border-[#3a3a3a] w-full flex-wrap text-xs sm:text-sm">
            <span className="text-[#999999] font-semibold mr-1">Устройства в комнате:</span>
            {peers.map((p) => {
              const isMe = p.id === myPeerId;
              const connType = peerConnectionTypes?.[p.id] || 'direct';
              return (
                <span
                  key={p.id}
                  className={`inline-flex items-center gap-1.5 sm:gap-2 px-2.5 sm:px-3 py-1 sm:py-1.5 rounded-lg border font-bold ${
                    isMe
                      ? 'bg-[#171717] text-[#ededed] border-[#3a3a3a]'
                      : 'bg-[#2d2d2d] text-[#ededed] border-[#3a3a3a]'
                  }`}
                >
                  <span className="w-2 h-2 rounded-full bg-[#789d2a]" />
                  <span>{p.name}</span>
                  {!isMe && (
                    <span
                      className={`text-[10px] px-1.5 py-0.2 rounded font-mono font-bold flex items-center gap-1 ${
                        connType === 'relay'
                          ? 'bg-amber-500/20 text-amber-300 border border-amber-500/40'
                          : 'bg-emerald-500/20 text-emerald-300 border border-emerald-500/40'
                      }`}
                    >
                      {connType === 'relay' ? '🛡️ TURN Relay' : '⚡ Прямой P2P'}
                    </span>
                  )}
                  {isMe && <span className="text-xs text-[#999999] font-normal">(Вы)</span>}
                </span>
              );
            })}
          </div>
        )}
      </div>

      {/* Pending Share Target Banner */}
      {((pendingSharedFiles && pendingSharedFiles.length > 0) || pendingSharedText) && (
        <div className="mb-5 p-4 rounded-xl bg-[#1c2214] border-2 border-[#789d2a] flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 shadow-md">
          <div className="text-sm text-[#ededed] min-w-0">
            <p className="font-bold text-[#bbf348]">
              Готово к отправке из системного меню «Поделиться»:
            </p>
            <p className="text-xs sm:text-sm text-[#cccccc] truncate">
              {pendingSharedFiles && pendingSharedFiles.length > 0
                ? `Файлов: ${pendingSharedFiles.length} (${pendingSharedFiles
                    .map((f) => f.name)
                    .join(', ')})`
                : pendingSharedText}
            </p>
          </div>
          <div className="flex items-center gap-2 shrink-0">
            <button
              type="button"
              onClick={onSendSharedBundle}
              disabled={otherPeers.length === 0}
              className="px-4 py-2 rounded-lg bg-[#789d2a] hover:bg-[#8cb632] disabled:opacity-50 text-white text-xs sm:text-sm font-bold cursor-pointer"
            >
              {otherPeers.length === 0 ? 'Ждём собеседника...' : 'Отправить сейчас'}
            </button>
            <button
              type="button"
              onClick={onDismissSharedBundle}
              className="px-3 py-2 rounded-lg bg-[#2d2d2d] hover:bg-[#3a3a3a] text-[#aaaaaa] text-xs sm:text-sm font-bold cursor-pointer"
            >
              Скрыть
            </button>
          </div>
        </div>
      )}

      {/* Waiting for peer banner */}
      {otherPeers.length === 0 && (
        <div className="mb-5 sm:mb-6 p-4 sm:p-5 rounded-2xl bg-gradient-to-r from-[#1d2616] via-[#202020] to-[#1a1a1a] border border-[#789d2a]/50 flex flex-col sm:flex-row items-center justify-between gap-4 sm:gap-5 shadow-lg">
          <div className="flex items-start gap-3 sm:gap-4 min-w-0 w-full sm:w-auto">
            <div className="w-3 h-3 sm:w-3.5 sm:h-3.5 rounded-full bg-[#789d2a] animate-ping shrink-0 mt-1.5" />
            <div className="space-y-1.5 min-w-0">
              <div className="flex items-center gap-2 flex-wrap">
                <p className="text-base sm:text-lg font-bold text-[#ededed]">
                  Ожидание второго устройства...
                </p>
                <span className="px-2.5 py-0.5 rounded-md bg-[#141414] border border-[#789d2a]/50 text-amber-300 font-mono text-sm sm:text-base font-black tracking-widest shadow-inner">
                  КОД: {pin}
                </span>
              </div>
              <p className="text-xs sm:text-sm text-[#aaaaaa] leading-relaxed">
                <span className="hidden sm:inline">
                  Наведите камеру смартфона на QR справа, продиктуйте 4-значный код или отправьте ссылку.
                </span>
                <span className="sm:hidden">
                  Продиктуйте 4-значный код, отправьте ссылку или покажите QR-код ниже.
                </span>
              </p>
              <div className="flex items-center gap-2 pt-1 flex-wrap">
                {supportsNativeShare && (
                  <button
                    onClick={handleNativeShare}
                    className="sm:hidden px-3.5 py-1.5 rounded-lg bg-[#789d2a] hover:bg-[#8cb632] text-white text-xs font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
                  >
                    <Share2 className="w-3.5 h-3.5" />
                    <span>Поделиться</span>
                  </button>
                )}
                <button
                  onClick={handleCopyLink}
                  className="px-3.5 py-1.5 rounded-lg bg-[#272727] hover:bg-[#333333] border border-[#3a3a3a] text-xs font-bold text-[#ededed] hover:text-white flex items-center gap-1.5 transition-colors cursor-pointer"
                >
                  {copiedLink ? (
                    <Check className="w-3.5 h-3.5 text-[#9ac43e]" />
                  ) : (
                    <Copy className="w-3.5 h-3.5" />
                  )}
                  <span>{copiedLink ? 'Ссылка скопирована' : 'Скопировать ссылку'}</span>
                </button>
                <button
                  onClick={onOpenQR}
                  className="hidden sm:flex px-3.5 py-1.5 rounded-lg bg-[#272727] hover:bg-[#333333] border border-[#3a3a3a] text-xs font-bold text-[#9ac43e] hover:text-[#bbf348] items-center gap-1.5 transition-colors cursor-pointer"
                >
                  <QrCode className="w-3.5 h-3.5" />
                  <span>Увеличить QR</span>
                </button>
              </div>
            </div>
          </div>

          <div
            onClick={onOpenQR}
            className="flex flex-col items-center gap-1 p-2 bg-white rounded-xl shadow-xl hover:scale-105 active:scale-95 transition-transform cursor-pointer shrink-0 border-2 border-white/20 group"
            title="Кликните, чтобы развернуть во весь экран"
          >
            <QRCodeSVG value={roomUrl} size={84} level="M" />
            <span className="text-[10px] text-gray-800 font-bold tracking-tight select-none group-hover:text-[#789d2a]">
              <span className="hidden sm:inline">Навести камеру</span>
              <span className="sm:hidden">Увеличить QR</span>
            </span>
          </div>
        </div>
      )}

      {/* Main Grid */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 sm:gap-6">
        <div className="lg:col-span-7 space-y-5">
          <div
            onDragOver={(e) => {
              e.preventDefault();
              setIsDragging(true);
            }}
            onDragLeave={() => setIsDragging(false)}
            onDrop={handleDrop}
            className={`rounded-xl border transition-all p-5 sm:p-8 text-center select-none shadow-md ${
              isDragging
                ? 'border-[#789d2a] bg-[#1c2214] ring-2 ring-[#789d2a]/30'
                : 'border-[#3a3a3a] bg-[#202020] hover:border-[#4a4a4a]'
            }`}
          >
            <input
              type="file"
              multiple
              ref={fileInputRef}
              onChange={(e) => {
                if (e.target.files) {
                  onSendFiles(e.target.files);
                  e.target.value = '';
                }
              }}
              className="hidden"
            />
            <input
              type="file"
              // @ts-ignore
              webkitdirectory=""
              directory=""
              multiple
              ref={folderInputRef}
              onChange={(e) => {
                if (e.target.files) {
                  onSendFiles(e.target.files);
                  e.target.value = '';
                }
              }}
              className="hidden"
            />

            <div className="hidden sm:flex items-center justify-center gap-2 mb-4">
              <div className="inline-flex items-center gap-2 px-3.5 py-1.5 rounded-lg bg-[#171717] border border-[#3a3a3a] text-sm">
                <kbd className="px-2 py-0.5 rounded bg-[#2a2a2a] text-[#ededed] font-mono font-bold border border-[#444444] text-xs shadow-inner">
                  Ctrl
                </kbd>
                <span className="text-[#999999] font-bold">+</span>
                <kbd className="px-2 py-0.5 rounded bg-[#2a2a2a] text-[#ededed] font-mono font-bold border border-[#444444] text-xs shadow-inner">
                  V
                </kbd>
                <span className="text-[#9ac43e] font-bold ml-1 text-sm sm:text-base">
                  вставить в любом месте
                </span>
              </div>
            </div>

            <h3 className="text-xl sm:text-3xl font-extrabold text-[#ededed] mb-2 tracking-tight">
              Передача файлов и папок
            </h3>

            <p className="hidden sm:block text-base sm:text-lg text-[#ededed] max-w-lg mx-auto mb-6 leading-relaxed">
              Нажмите{' '}
              <strong className="text-white bg-[#171717] px-2 py-0.5 rounded border border-[#3a3a3a]">
                Ctrl + V
              </strong>{' '}
              для мгновенной отправки скриншота или файла, перетащите мышкой в окно или выберите вручную:
            </p>
            <p className="sm:hidden text-sm text-[#cccccc] max-w-md mx-auto mb-5 leading-relaxed">
              Выберите фото, видео или файлы с устройства — они передадутся напрямую без сжатия:
            </p>

            <div className="flex flex-wrap items-center justify-center gap-2.5 sm:gap-3">
              <button
                type="button"
                onClick={() => fileInputRef.current?.click()}
                className="px-5 py-2.5 rounded-lg bg-[#789d2a] hover:bg-[#8cb632] text-white font-bold text-sm sm:text-base flex items-center gap-2 transition-colors cursor-pointer shadow-sm"
              >
                <File className="w-4 h-4 sm:w-5 sm:h-5" />
                <span>Выбрать файлы</span>
              </button>

              {!isIOS && (
                <button
                  type="button"
                  onClick={() => folderInputRef.current?.click()}
                  className="px-4 sm:px-5 py-2.5 rounded-lg bg-[#2d2d2d] hover:bg-[#3a3a3a] text-[#ededed] font-bold text-sm sm:text-base flex items-center gap-2 transition-colors cursor-pointer border border-[#3a3a3a]"
                >
                  <Folder className="w-4 h-4 sm:w-5 sm:h-5 text-[#9ac43e]" />
                  <span>Выбрать папку</span>
                </button>
              )}

              <button
                type="button"
                onClick={handlePasteFromClipboard}
                className="px-4 sm:px-5 py-2.5 rounded-lg bg-[#2d2d2d] hover:bg-[#3a3a3a] text-[#ededed] font-bold text-sm sm:text-base flex items-center gap-2 transition-colors cursor-pointer border border-[#3a3a3a]"
                title="Вставить из буфера обмена (Ctrl + V)"
              >
                <Copy className="w-4 h-4 sm:w-5 sm:h-5 text-[#aaaaaa]" />
                <span>Из буфера</span>
              </button>
            </div>

            <div className="flex flex-wrap items-center justify-center gap-2 sm:gap-4 pt-4 mt-5 sm:mt-6 border-t border-[#3a3a3a]/60 text-xs sm:text-base text-[#aaaaaa]">
              <span className="flex items-center gap-1.5">
                <span className="w-2 h-2 rounded-full bg-[#789d2a]" />
                Прямой WebRTC канал
              </span>
              <span className="text-[#555555]">·</span>
              <span>Без ограничений по размеру</span>
              <span className="text-[#555555]">·</span>
              <span className="hidden sm:inline">Drag & Drop в любую область</span>
              <span className="sm:hidden">Без сжатия качества</span>
            </div>
          </div>

          {transfers.length > 0 && (
            <div className="bg-[#222222] border border-[#3a3a3a] rounded-xl p-4 sm:p-6 shadow-sm">
              <div className="flex flex-wrap items-center justify-between gap-2 mb-3.5 sm:mb-4">
                <h4 className="text-base sm:text-xl font-bold text-[#ededed]">
                  Передачи файлов ({transfers.length})
                </h4>

                {downloadableTransfers.length >= 2 && (
                  <button
                    type="button"
                    onClick={handleDownloadAllZip}

                    className="px-3.5 py-1.5 rounded-lg bg-[#789d2a] hover:bg-[#8cb632] disabled:opacity-60 text-white text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer shadow-sm"
                  >
                    <Package className="w-4 h-4" />
                    <span>
                      {isZipping
                        ? 'Собираем ZIP...'
                        : `Скачать всё ZIP (${downloadableTransfers.length})`}
                    </span>
                  </button>
                )}
              </div>

              <div className="space-y-3">
                {transfers.map((t) => {
                  const timing = transferTimingRef.current[t.id];
                  if (t.status === 'transferring' && !timing) {
                    transferTimingRef.current[t.id] = { start: Date.now() };
                  } else if (
                    t.status === 'completed' &&
                    timing &&
                    timing.durationSec === undefined
                  ) {
                    const durationSec = Math.max(0.1, (Date.now() - timing.start) / 1000);
                    timing.durationSec = durationSec;
                    timing.avgSpeed = t.size > 0 ? t.size / durationSec : 0;
                  }
                  const completedTiming = transferTimingRef.current[t.id];

                  const isImage =
                    t.mimeType.startsWith('image/') ||
                    /\.(png|jpe?g|gif|webp|svg|bmp)$/i.test(t.name);
                  const isVideo =
                    t.mimeType.startsWith('video/') ||
                    /\.(mp4|webm|mov|mkv|avi)$/i.test(t.name);
                  const isAudio =
                    t.mimeType.startsWith('audio/') ||
                    /\.(mp3|wav|ogg|m4a|aac)$/i.test(t.name);

                  return (
                    <div
                      key={t.id}
                      className="p-3.5 sm:p-4 rounded-lg bg-[#171717] border border-[#3a3a3a] flex flex-col gap-3 transition-all"

                                alt={t.name}

                            </div>

                              onClick={() => {

                              }}
                              className="relative group shrink-0 w-12 h-12 sm:w-13 sm:h-13 rounded-lg bg-[#151522] border border-[#3a3a3a] group-hover:border-[#789d2a] flex items-center justify-center cursor-pointer transition-all"

                            >
                              <Video className="w-6 h-6 text-[#9ac43e]" />
                              <div className="absolute inset-0 bg-black/50 rounded-lg opacity-0 group-hover:opacity-100 flex items-center justify-center transition-opacity text-white">
                                <Eye className="w-4 h-4" />
                              </div>
                            </div>
                          ) : (
                            <div className="p-2.5 rounded-lg bg-[#222222] border border-[#3a3a3a] shrink-0">
                              {getFileIcon(t.mimeType)}
                            </div>
                          )}

                          <div className="min-w-0">
                            <p className="text-sm sm:text-lg font-bold text-[#ededed] truncate max-w-[160px] sm:max-w-md">
                              {t.name}
                            </p>
                            <p className="text-xs sm:text-sm text-[#aaaaaa] flex items-center flex-wrap gap-1.5 sm:gap-2 mt-0.5">
                              <span>{formatBytes(t.size)}</span>
                              <span>•</span>
                              <span
                                className={
                                  t.type === 'upload'
                                    ? 'text-[#9ac43e] font-semibold'
                                    : 'text-[#ededed]'
                                }
                              >
                                {t.type === 'upload' ? 'Отправка' : 'Получение'}
                              </span>
                              {t.status === 'transferring' && t.speed > 0 && (
                                <>
                                  <span>•</span>
                                  <span className="font-mono text-[#9ac43e] font-bold">
                                    {formatSpeed(t.speed)}
                                  </span>
                                </>
                              )}
                              {t.status === 'completed' &&
                                completedTiming?.durationSec !== undefined &&
                                (completedTiming.avgSpeed || 0) > 0 && (
                                  <>
                                    <span>•</span>
                                    <span className="font-mono text-[#9ac43e] text-xs">
                                      {completedTiming.durationSec.toFixed(1)} сек ·{' '}
                                      {formatSpeed(completedTiming.avgSpeed!)}
                                    </span>
                                  </>
                                )}
                              {!t.savedToDisk && t.size > 1024 * 1024 * 1024 && (
                                <>
                                  <span>•</span>
                                  <span className="text-amber-400 font-mono text-xs">
                                    Память браузера
                                  </span>
                                </>
                              )}
                            </p>
                          </div>
                        </div>

                        <div className="flex items-center gap-1.5 sm:gap-2 shrink-0">
                          {t.status === 'completed' && t.savedToDisk && (
                            <span className="px-2.5 sm:px-3 py-1.5 rounded-lg bg-[#789d2a]/15 border border-[#789d2a]/40 text-[#9ac43e] text-xs sm:text-sm font-bold flex items-center gap-1.5">
                              <Check className="w-4 h-4" />
                              <span className="hidden sm:inline">Сохранено на диск</span>
                            </span>
                          )}

                          {t.blobUrl && (isImage || isVideo || isAudio) && (
                            <button
                              type="button"
                              onClick={() =>
                                setPreviewMedia({
                                  url: t.blobUrl!,
                                  name: t.name,
                                  type: isImage ? 'image' : isVideo ? 'video' : 'audio',
                                })
                              }
                              className="p-2 rounded-lg bg-[#222222] hover:bg-[#2d2d2d] text-[#aaaaaa] hover:text-[#9ac43e] border border-[#3a3a3a] transition-colors cursor-pointer"
                              title="Предпросмотр"
                            >
                              <Eye className="w-4 h-4" />
                            </button>
                          )}

                          {t.status === 'completed' && !t.savedToDisk && t.blobUrl && (
                            <a
                              href={t.blobUrl}
                              download={t.name}
                              className="px-3 sm:px-4 py-2 rounded-lg bg-[#789d2a] hover:bg-[#8cb632] text-white text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors shadow-sm"
                            >
                              <Download className="w-4 h-4" />
                              <span className="hidden sm:inline">Скачать</span>
                            </a>
                          )}

                          {t.status === 'transferring' && (
                            <button
                              onClick={() => onCancelTransfer(t.id)}
                              className="p-2 rounded-lg hover:bg-[#2d2d2d] text-[#aaaaaa] hover:text-[#ef9282] transition-colors cursor-pointer"
                              title="Отменить"
                            >
                              <X className="w-4 h-4" />
                            </button>
                          )}

                          {t.status === 'error' && (
                            <span className="text-xs sm:text-sm text-[#ef9282] flex items-center gap-1 font-semibold">
                              <AlertCircle className="w-4 h-4" />
                              <span>{t.error || 'Ошибка'}</span>
                            </span>
                          )}
                        </div>
                      </div>

                      {isAudio && t.blobUrl && (
                        <div className="mt-1 pt-2 border-t border-[#222222]">
                          <CustomAudioPlayer src={t.blobUrl} compact title={t.name} />
                        </div>
                      )}

                      {t.status === 'pending_approval' && (
                        <div className="mt-1 p-3.5 rounded-lg bg-[#222222] border border-[#3a3a3a] flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                          <div className="flex items-center gap-2.5 text-xs sm:text-sm text-[#ededed]">
                            <HardDrive className="w-5 h-5 text-[#9ac43e] shrink-0" />
                            <span>
                              Большой файл. Выберите место сохранения на диске без расхода RAM.
                            </span>
                          </div>
                          <div className="flex items-center gap-2 shrink-0">
                            <button
                              onClick={() => onAcceptFile(t.id)}
                              className="px-3.5 py-2 rounded-lg bg-[#789d2a] hover:bg-[#8cb632] text-white text-xs sm:text-sm font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
                            >
                              <FolderDown className="w-4 h-4" />
                              <span>Принять</span>
                            </button>
                            <button
                              onClick={() => onRejectFile(t.id)}
                              className="px-3 py-2 rounded-lg bg-[#2d2d2d] hover:bg-[#3a3a3a] text-[#ededed] text-xs sm:text-sm font-bold transition-colors cursor-pointer"
                            >
                              Отклонить
                            </button>
                          </div>
                        </div>
                      )}

                      {t.status === 'waiting_receiver' && (
                        <div className="mt-1 text-xs sm:text-sm text-[#9ac43e] flex items-center gap-2 font-semibold">
                          <span className="w-2.5 h-2.5 rounded-full bg-[#789d2a] animate-pulse shrink-0" />
                          <span>Ожидание: получатель подтверждает сохранение...</span>
                        </div>
                      )}

                      {t.status === 'transferring' && (
                        <div className="w-full bg-[#222222] rounded-full h-2 overflow-hidden border border-[#3a3a3a]/40">
                          <div
                            className="bg-[#789d2a] h-2 rounded-full transition-all duration-300"
                            style={{ width: `${t.progress}%` }}
                          />
                        </div>
                      )}
                    </div>
                  );
                })}
              </div>
            </div>
          )}
        </div>

        {/* Right Column: Clipboard & Text */}
        <div className="lg:col-span-5 flex flex-col h-[460px] lg:h-[580px] bg-[#222222] border border-[#3a3a3a] rounded-xl p-4 sm:p-6 shadow-sm">
          <div className="flex items-center justify-between gap-2 mb-3 shrink-0">
            <div className="min-w-0">
              <h4 className="text-base sm:text-lg font-bold text-[#ededed] truncate">
                Мгновенный буфер и текст
              </h4>
              <p className="text-xs sm:text-sm text-[#aaaaaa] truncate">
                Ссылки, пароли, заметки в 1 клик
              </p>
            </div>
            <button
              onClick={handlePasteFromClipboard}
              className="px-2.5 sm:px-3 py-1.5 rounded-lg bg-[#2d2d2d] hover:bg-[#3a3a3a] text-[#ededed] text-xs font-bold flex items-center gap-1.5 transition-colors cursor-pointer border border-[#3a3a3a] shrink-0"
              title="Вставить из системного буфера"
            >
              <Copy className="w-3.5 h-3.5 text-[#9ac43e]" />
              <span className="hidden sm:inline">Вставить из буфера</span>
              <span className="sm:hidden">Буфер</span>
            </button>
          </div>

          <div className="flex-1 overflow-y-auto space-y-3.5 pr-1 py-1">
            {messages.length === 0 ? (
              <div className="h-full flex flex-col items-center justify-center text-center p-6 text-[#aaaaaa]">
                <FileText className="w-10 h-10 stroke-1 mb-2.5 opacity-40" />
                <p className="text-base font-semibold text-[#ededed]">
                  Вставьте ссылку или текст ниже
                </p>
                <p className="text-xs sm:text-sm text-[#aaaaaa] mt-1">
                  У собеседника моментально появится кнопка «Скопировать»
                </p>
              </div>
            ) : (
              messages.map((m) => {
                const isMe = m.senderId === myPeerId;
                const peer = peers.find((p) => p.id === m.senderId);
                const currentAvatar = isMe
                  ? userAvatar || getAvatarForName(m.senderName)
                  : peer?.avatar || m.senderAvatar || getAvatarForName(m.senderName);
                const messageIsAdmin = isMe ? isAdmin : Boolean(peer?.isAdmin || m.isAdmin);
                return (
                  <div
                    key={m.id}
                    className={`p-3.5 rounded-lg border group ${
                      messageIsAdmin
                        ? 'bg-gradient-to-br from-[#22180b] to-[#161616] border-amber-500/50 shadow-[0_0_12px_rgba(245,158,11,0.15)]'
                        : 'bg-[#171717] border-[#3a3a3a]'
                    }`}
                  >
                    <div className="flex items-center justify-between text-xs text-[#aaaaaa] mb-2 font-semibold">
                      <div className="flex items-center gap-2 min-w-0">
                        <img
                          src={currentAvatar}
                          alt={m.senderName}
                          className={`w-5 h-5 rounded-full object-cover bg-[#222] shrink-0 ${
                            messageIsAdmin ? 'border border-amber-400' : 'border border-[#789d2a]/50'
                          }`}
                          style={{ imageRendering: 'pixelated' }}
                        />
                        <div className="flex items-center gap-1.5 min-w-0">
                          {messageIsAdmin && (
                            <span className="px-1 py-0.2 rounded text-[9px] font-mono font-black bg-amber-400/20 text-amber-300 border border-amber-400/40 uppercase tracking-wider select-none shrink-0">
                              DEV
                            </span>
                          )}
                          <span
                            className={`font-bold truncate ${
                              messageIsAdmin ? 'admin-name-shimmer text-amber-300' : 'text-[#ededed]'
                            }`}
                          >
                            {m.senderName}
                          </span>
                          {(m.senderStatusEmote ||
                            (isMe ? statusEmote || undefined : peer?.statusEmote)) && (
                            <img
                              src={
                                m.senderStatusEmote ||
                                (isMe ? statusEmote || undefined : peer?.statusEmote)
                              }
                              alt="status"
                              className="w-4 h-4 object-contain inline-block select-none filter drop-shadow-[0_0_4px_rgba(255,255,255,0.3)] shrink-0"
                            />
                          )}
                        </div>
                      </div>
                      <span className="shrink-0 ml-2">
                        {new Date(m.timestamp).toLocaleTimeString([], {
                          hour: '2-digit',
                          minute: '2-digit',
                        })}
                      </span>
                    </div>

                    <p className="text-base sm:text-lg text-[#ededed] break-words whitespace-pre-wrap select-all font-sans leading-relaxed">
                      {m.text}
                    </p>

                    <div className="mt-3 pt-2.5 border-t border-[#222222] flex items-center justify-end gap-2">
                      {isUrl(m.text) && (
                        <a
                          href={m.text}
                          target="_blank"
                          rel="noreferrer"
                          className="px-3 py-1.5 rounded-lg bg-[#222222] hover:bg-[#2d2d2d] text-xs font-bold text-[#9ac43e] flex items-center gap-1.5 transition-colors"
                        >
                          <ExternalLink className="w-3.5 h-3.5" />
                          <span>Открыть</span>
                        </a>
                      )}
                      <button
                        onClick={() => handleCopyText(m.id, m.text)}
                        className="px-3 py-1.5 rounded-lg bg-[#222222] hover:bg-[#2d2d2d] text-xs font-bold text-[#ededed] flex items-center gap-1.5 transition-colors cursor-pointer"
                      >
                        {copiedTextId === m.id ? (
                          <>
                            <Check className="w-3.5 h-3.5 text-[#9ac43e]" />
                            <span className="text-[#9ac43e]">Скопировано!</span>
                          </>
                        ) : (
                          <>
                            <Copy className="w-3.5 h-3.5" />
                            <span>Скопировать</span>
                          </>
                        )}
                      </button>
                    </div>
                  </div>
                );
              })
            )}
            <div ref={chatBottomRef} />
          </div>

          <form
            onSubmit={handleTextSubmit}
            className="mt-3 pt-3 border-t border-[#3a3a3a] shrink-0"
          >
            <div className="flex items-center gap-2">
              <img
                src={
                  userAvatar || getAvatarForName(peers.find((p) => p.id === myPeerId)?.name || 'Вы')
                }
                alt="Ваш аватар"
                className="w-8 h-8 sm:w-10 sm:h-10 rounded-full object-cover border-2 border-[#789d2a] shadow-[0_0_8px_rgba(120,157,42,0.3)] bg-[#171717] shrink-0"
                style={{ imageRendering: 'pixelated' }}
                title="Ваш аватар"
              />
              <input
                type="text"
                value={textInput}
                onChange={(e) => setTextInput(e.target.value)}
                placeholder="Вставьте ссылку или текст..."
                className="flex-1 min-w-0 w-full bg-[#171717] border border-[#3a3a3a] rounded-lg px-3 sm:px-4 py-2 sm:py-2.5 text-sm sm:text-base text-[#ededed] placeholder-[#999999] outline-none focus:border-[#789d2a] transition-colors"
              />
              <button
                type="submit"
                disabled={!textInput.trim()}
                className="px-3.5 sm:px-5 py-2 sm:py-2.5 rounded-lg bg-[#789d2a] hover:bg-[#8cb632] text-white font-bold text-sm sm:text-base transition-colors disabled:opacity-40 cursor-pointer shrink-0"
              >
                Отправить
              </button>
            </div>
          </form>
        </div>
      </div>

      {/* Media Lightbox */}
      {previewMedia && (
        <div
          className="fixed inset-0 z-50 bg-black/90 backdrop-blur-sm flex flex-col items-center justify-center p-4 animate-in fade-in"
          onClick={() => setPreviewMedia(null)}
        >
          <div
            className="relative max-w-4xl w-full flex flex-col items-center"
            onClick={(e) => e.stopPropagation()}
          >
            <div className="w-full flex items-center justify-between pb-3 text-base text-[#ededed]">
              <span className="font-bold truncate max-w-xs sm:max-w-md">{previewMedia.name}</span>
              <div className="flex items-center gap-2 shrink-0">
                <a
                  href={previewMedia.url}
                  download={previewMedia.name}
                  className="px-3.5 sm:px-4 py-2 rounded-lg bg-[#789d2a] text-white text-xs sm:text-sm font-bold hover:bg-[#8cb632] transition-colors flex items-center gap-1.5 shadow-md"
                >
                  <Download className="w-4 h-4" />
                  <span>Скачать</span>
                </a>
                <button
                  onClick={() => setPreviewMedia(null)}
                  className="p-2 rounded-lg bg-[#222222] border border-[#3a3a3a] text-[#aaaaaa] hover:text-white transition-colors cursor-pointer"
                >
                  <X className="w-4 h-4" />
                </button>
              </div>
            </div>
            {previewMedia.type === 'image' && (
              <img
                src={previewMedia.url}
                alt={previewMedia.name}
                className="max-h-[80vh] max-w-full object-contain rounded-lg border border-[#3a3a3a] shadow-2xl bg-[#171717]"
              />
            )}
            {previewMedia.type === 'video' && (
              <CustomVideoPlayer
                src={previewMedia.url}
                title={previewMedia.name}
                autoPlay
                className="w-full"
                onDownload={() => {
                  const a = document.createElement('a');
                  a.href = previewMedia.url;
                  a.download = previewMedia.name;
                  document.body.appendChild(a);
                  a.click();
                  document.body.removeChild(a);
                }}
              />
            )}
            {previewMedia.type === 'audio' && (
              <div className="p-6 rounded-2xl bg-[#1c1c24] border border-[#3a3a45] w-full max-w-lg">
                <CustomAudioPlayer src={previewMedia.url} title={previewMedia.name} autoPlay />
              </div>
            )}
          </div>
        </div>
      )}

      {toastMessage && (
        <div className="fixed bottom-6 right-4 sm:right-6 z-50 px-4 sm:px-5 py-3 rounded-xl bg-[#202020] border border-[#789d2a] text-[#ededed] text-xs sm:text-base font-bold shadow-2xl flex items-center gap-2.5 animate-in slide-in-from-bottom-2 max-w-[90vw]">
          <span className="w-2.5 h-2.5 rounded-full bg-[#789d2a] shrink-0" />
          <span>{toastMessage}</span>
        </div>
      )}
    </div>
  );
};
